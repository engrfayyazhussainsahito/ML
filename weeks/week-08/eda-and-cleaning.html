Lesson 8.2: EDA and Data Cleaning

The Goal of EDA

Exploratory Data Analysis is a systematic investigation of a dataset before modelling. The goal is to understand distributions, detect data quality issues, discover relationships between features and the target, and generate hypotheses that guide feature engineering. EDA is always done on the training set only — never on test data.

Univariate Analysis

import pandas as pd
import numpy as np

df = pd.read_csv("train.csv")

# Numerical features
print("=== Numerical Summary ===")
print(df.describe().T.round(3))

# Check skewness (|skew| > 1 suggests log transform)
print("\nSkewness:")
print(df.select_dtypes(np.number).skew().sort_values(key=abs, ascending=False))

# Categorical features
print("\n=== Categorical Cardinality ===")
cat_cols = df.select_dtypes('object').columns
for col in cat_cols:
    n_unique = df[col].nunique()
    top = df[col].value_counts().index[0]
    pct = df[col].value_counts(normalize=True).iloc[0]
    print(f"  {col}: {n_unique} unique, most common='{top}' ({pct:.1%})")

Bivariate Analysis

import pandas as pd
import numpy as np

target = 'Survived'

# Numerical vs target: means by class, t-test
from scipy import stats
num_cols = df.select_dtypes(np.number).columns.drop(target, errors='ignore')

print("=== Numerical vs Target ===")
for col in num_cols:
    pos = df.loc[df[target]==1, col].dropna()
    neg = df.loc[df[target]==0, col].dropna()
    t_stat, p_val = stats.ttest_ind(pos, neg)
    print(f"  {col}: mean(pos)={pos.mean():.2f}, mean(neg)={neg.mean():.2f}, p={p_val:.4f}")

# Correlation matrix (numerical only)
corr = df[num_cols].corr()
print("\n=== High Correlations (|r| > 0.7) ===")
for i in range(len(corr)):
    for j in range(i+1, len(corr)):
        r = corr.iloc[i, j]
        if abs(r) > 0.7:
            print(f"  {corr.index[i]} vs {corr.columns[j]}: r={r:.3f}")

Data Quality Audit

import pandas as pd, numpy as np

def data_quality_report(df):
    report = pd.DataFrame({
        'dtype':    df.dtypes,
        'n_null':   df.isnull().sum(),
        'pct_null': df.isnull().mean() * 100,
        'n_unique': df.nunique(),
        'n_zeros':  (df == 0).sum(),
    })

    # Flag suspicious columns
    report['flags'] = ''
    report.loc[report['pct_null'] > 50, 'flags'] += '|high_missing'
    report.loc[report['n_unique'] == 1,  'flags'] += '|constant'
    report.loc[report['n_unique'] == len(df), 'flags'] += '|possible_ID'

    return report

report = data_quality_report(df)
print(report[report['flags'] != ''])

# Duplicate rows
dupes = df.duplicated().sum()
print(f"\nDuplicate rows: {dupes} ({dupes/len(df):.1%})")

# Outliers (IQR method)
for col in df.select_dtypes(np.number).columns:
    Q1, Q3 = df[col].quantile([0.25, 0.75])
    IQR = Q3 - Q1
    n_outliers = ((df[col] < Q1 - 3*IQR) | (df[col] > Q3 + 3*IQR)).sum()
    if n_outliers > 0:
        print(f"  {col}: {n_outliers} extreme outliers")
