Lesson 5.2: Boosting and XGBoost

Boosting: Sequential Correction

Where bagging trains trees independently and averages, boosting trains trees sequentially — each new tree corrects the errors of the previous ensemble. The result is a strong learner built from many weak ones. Boosting consistently dominates structured/tabular data competitions.

Gradient Boosting Machines

GBM frames boosting as gradient descent in function space. At each stage we fit a new tree to the negative gradient of the loss with respect to current predictions — for MSE regression this is the residual; for log-loss classification it is a different form. Each tree is a small gradient descent step.

from sklearn.ensemble import GradientBoostingClassifier
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
import numpy as np

X, y = make_classification(n_samples=2000, n_features=20, random_state=42)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)

gbm = GradientBoostingClassifier(
    n_estimators=400,
    learning_rate=0.05,   # shrinkage: each tree contributes less
    max_depth=4,
    subsample=0.8,        # stochastic gradient boosting
    min_samples_leaf=5,
    random_state=42
)
gbm.fit(X_train, y_train)
print(f"AUC: {roc_auc_score(y_test, gbm.predict_proba(X_test)[:,1]):.4f}")

# Staged prediction: AUC as a function of number of trees
aucs = [roc_auc_score(y_test, proba[:,1])
        for proba in gbm.staged_predict_proba(X_test)]
best_n = np.argmax(aucs) + 1
print(f"Best n_estimators: {best_n}, AUC={aucs[best_n-1]:.4f}")

XGBoost

XGBoost (Chen & Guestrin, 2016) extends GBM with a second-order Taylor expansion of the loss, explicit regularisation on tree complexity (L1/L2 on leaf weights), and column/row subsampling. It is typically 10–100× faster than sklearn GBM and usually more accurate.

import xgboost as xgb
from sklearn.model_selection import cross_val_score

xgb_clf = xgb.XGBClassifier(
    n_estimators=500,
    learning_rate=0.05,
    max_depth=6,
    subsample=0.8,
    colsample_bytree=0.8,   # fraction of features per tree
    reg_alpha=0.1,          # L1 regularisation on leaf weights
    reg_lambda=1.0,         # L2 regularisation on leaf weights
    use_label_encoder=False,
    eval_metric='logloss',
    random_state=42,
    n_jobs=-1
)

scores = cross_val_score(xgb_clf, X, y, cv=5, scoring='roc_auc', n_jobs=-1)
print(f"XGBoost AUC: {scores.mean():.4f} +/- {scores.std():.4f}")

# Early stopping (use a validation set, not test!)
xgb_clf.fit(X_train, y_train,
            eval_set=[(X_test, y_test)],
            early_stopping_rounds=30,
            verbose=False)
print(f"Best iteration: {xgb_clf.best_iteration}")

LightGBM: Leaf-Wise Growth

import lightgbm as lgb
from sklearn.model_selection import cross_val_score

lgb_clf = lgb.LGBMClassifier(
    n_estimators=1000,
    learning_rate=0.03,
    num_leaves=63,          # controls model complexity (not max_depth)
    min_child_samples=20,
    subsample=0.8,
    colsample_bytree=0.8,
    reg_alpha=0.1,
    reg_lambda=0.1,
    random_state=42,
    n_jobs=-1,
    verbosity=-1
)
scores = cross_val_score(lgb_clf, X, y, cv=5, scoring='roc_auc', n_jobs=-1)
print(f"LightGBM AUC: {scores.mean():.4f} +/- {scores.std():.4f}")
