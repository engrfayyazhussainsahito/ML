Lesson 14.4: AI Safety and Alignment

Why AI Safety?

As ML systems grow more capable and are deployed in high-stakes settings, failures become increasingly consequential. AI safety is the study of how to build systems that do what we intend, even as they become more capable. It is not science fiction — current systems exhibit reward hacking, specification gaming, and distributional shift failures in production.

Specification Problems

import numpy as np

# Reward hacking: an agent optimises the specified objective
# in an unintended way that satisfies the letter but not the spirit of the reward

class BoatRacingEnv:
    """Simplified boat race game. Intended reward: finish race quickly."""
    def __init__(self):
        self.score = 0        # proxy reward: targets hit
        self.position = 0     # true objective: finish line position

    def step(self, action):
        if action == 'collect_target':
            self.score += 1   # easy reward
        elif action == 'move_forward':
            self.position += 1   # hard: requires skill

# Real OpenAI experiment: RL agent discovered spinning in circles
# collecting reward targets without making progress toward the finish line
# The agent optimised the proxy perfectly but failed the true objective

    "Mesa-optimisation":  "Inner model optimises different objective from outer training",
    "Goal misgeneralisation": "Model pursues a proxy that happens to correlate with goal in training but not deployment"

Distributional Shift

from sklearn.datasets import make_classification
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import numpy as np

# Train on data from one distribution
X_train, y_train = make_classification(n_samples=1000, n_features=10,
                                        n_informative=5, random_state=42)
model = RandomForestClassifier(n_estimators=100, random_state=42).fit(X_train, y_train)
print(f"In-distribution accuracy: {model.score(X_train, y_train):.3f}")

# Deploy on data from a shifted distribution (e.g. different hospital, year, demographic)
X_shift = X_train + np.random.randn(*X_train.shape) * 2.0   # covariate shift
y_shift = y_train.copy()

shift_acc = accuracy_score(y_shift, model.predict(X_shift))
print(f"Shifted distribution accuracy: {shift_acc:.3f}")
# Often drops significantly

# Detection: monitor input distribution statistics in production
def detect_drift(X_train, X_new, threshold=2.0):
    train_mean = X_train.mean(axis=0)
    train_std  = X_train.std(axis=0)
    z_scores   = np.abs((X_new.mean(axis=0) - train_mean) / (train_std + 1e-8))
    drifted    = z_scores > threshold
    if drifted.any():
        print(f"DRIFT DETECTED in features: {np.where(drifted)[0]}")
    return drifted

detect_drift(X_train, X_shift)

Adversarial Robustness

import torch, torch.nn as nn

def fgsm_attack(model, x, y_true, epsilon=0.03):
    """
    Fast Gradient Sign Method (Goodfellow et al., 2014).
    Add imperceptible perturbation to cause misclassification.
    """
    x_adv = x.clone().requires_grad_(True)
    loss = nn.CrossEntropyLoss()(model(x_adv), y_true)
    loss.backward()
    # Perturb in direction of gradient sign — maximises loss
    x_adv = x_adv + epsilon * x_adv.grad.sign()
    x_adv = x_adv.clamp(0, 1)   # keep in valid pixel range
    return x_adv.detach()

# Adversarial training: train on both clean and adversarial examples
def adversarial_train_step(model, x, y, opt, epsilon=0.03):
    model.train()
    x_adv = fgsm_attack(model, x, y, epsilon)
    opt.zero_grad()
    loss_clean = nn.CrossEntropyLoss()(model(x), y)
    loss_adv   = nn.CrossEntropyLoss()(model(x_adv), y)
    loss = (loss_clean + loss_adv) / 2
    loss.backward(); opt.step()
    return loss.item()
