Lesson 9.2: Feedforward Networks

Feedforward Network Architecture

A feedforward network (multilayer perceptron, MLP) stacks multiple layers of neurons. Each layer transforms its input with a linear projection followed by a non-linear activation. Information flows in one direction: input → hidden layers → output. The Universal Approximation Theorem (Hornik, 1989) guarantees that a sufficiently wide single hidden layer can approximate any continuous function on a compact set — but in practice, depth is more efficient than width.

Building MLPs in PyTorch

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import TensorDataset, DataLoader
import numpy as np

class MLP(nn.Module):
    def __init__(self, input_dim, hidden_dims, output_dim, dropout=0.3):
        super().__init__()
        layers = []
        prev_dim = input_dim
        for dim in hidden_dims:
            layers += [
                nn.Linear(prev_dim, dim),
                nn.BatchNorm1d(dim),
                nn.GELU(),
                nn.Dropout(dropout)
            ]
            prev_dim = dim
        layers.append(nn.Linear(prev_dim, output_dim))
        self.net = nn.Sequential(*layers)

    def forward(self, x):
        return self.net(x)

# Instantiate
model = MLP(
    input_dim=20,
    hidden_dims=[256, 128, 64],
    output_dim=1,
    dropout=0.3
)
total_params = sum(p.numel() for p in model.parameters())
print(f"Parameters: {total_params:,}")
print(model)

The Training Loop

from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import torch, numpy as np

# Data
X_np, y_np = make_classification(n_samples=5000, n_features=20, random_state=42)
X_tr, X_val, y_tr, y_val = train_test_split(X_np, y_np, test_size=0.2)
sc = StandardScaler(); X_tr = sc.fit_transform(X_tr); X_val = sc.transform(X_val)

def to_tensor(*arrays): return [torch.FloatTensor(a) for a in arrays]
X_tr_t, y_tr_t = to_tensor(X_tr, y_tr.reshape(-1,1))
X_val_t, y_val_t = to_tensor(X_val, y_val.reshape(-1,1))

loader = DataLoader(TensorDataset(X_tr_t, y_tr_t), batch_size=64, shuffle=True)
model = MLP(20, [128,64,32], 1); criterion = nn.BCEWithLogitsLoss()
opt = optim.Adam(model.parameters(), lr=1e-3, weight_decay=1e-4)
scheduler = optim.lr_scheduler.CosineAnnealingLR(opt, T_max=50)

best_val, patience_counter, best_state = float('inf'), 0, None
for epoch in range(200):
    model.train()
    for xb, yb in loader:
        opt.zero_grad()
        nn.BCEWithLogitsLoss()(model(xb), yb).backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        opt.step()
    scheduler.step()

    model.eval()
    with torch.no_grad():
        val_loss = nn.BCEWithLogitsLoss()(model(X_val_t), y_val_t).item()
        acc = ((model(X_val_t).sigmoid() > 0.5) == y_val_t).float().mean().item()

    if val_loss < best_val:
        best_val = val_loss; patience_counter = 0
        best_state = {k: v.clone() for k, v in model.state_dict().items()}
    else:
        patience_counter += 1
        if patience_counter >= 20:
            print(f"Early stop at epoch {epoch+1}"); break

    if epoch % 20 == 0:
        print(f"Epoch {epoch+1:3d}: val_loss={val_loss:.4f}, acc={acc:.4f}")

model.load_state_dict(best_state)

Weight Initialisation

import torch.nn as nn

# Poor initialisation -> vanishing or exploding gradients at start
# Kaiming (He) init: designed for ReLU — var(w) = 2/fan_in
# Xavier (Glorot) init: designed for sigmoid/tanh — var(w) = 2/(fan_in+fan_out)

def init_weights(module):
    if isinstance(module, nn.Linear):
        nn.init.kaiming_normal_(module.weight, nonlinearity='relu')
        nn.init.zeros_(module.bias)

model = MLP(20, [128,64,32], 1)
model.apply(init_weights)

# Verify initial gradient magnitudes are reasonable
x = torch.randn(32, 20)
y = torch.randint(0, 2, (32, 1)).float()
loss = nn.BCEWithLogitsLoss()(model(x), y)
loss.backward()
for name, p in model.named_parameters():
    if p.grad is not None:
        print(f"{name}: grad_norm = {p.grad.norm():.4f}")
