Lesson 12.3: BERT, GPT and Foundation Models

Pre-training Objectives

Large language models are pre-trained on massive text corpora with self-supervised objectives — the labels are derived from the data itself, requiring no human annotation. The objective shapes what the model learns: BERT learns bidirectional context; GPT learns autoregressive prediction.

BERT: Masked Language Modelling

from transformers import BertTokenizer, BertForMaskedLM
import torch

tokeniser = BertTokenizer.from_pretrained('bert-base-uncased')
model = BertForMaskedLM.from_pretrained('bert-base-uncased')
model.eval()

text = "The capital of France is [MASK]."
inputs = tokeniser(text, return_tensors='pt')
mask_idx = (inputs['input_ids'] == tokeniser.mask_token_id).nonzero()[0, 1]

with torch.no_grad():
    logits = model(**inputs).logits

# Top-5 predictions for the masked token
top_tokens = logits[0, mask_idx].topk(5)
for score, idx in zip(top_tokens.values, top_tokens.indices):
    print(f"  {tokeniser.decode([idx]):15s}: {score:.2f}")
# Expected: 'paris', 'lyon', etc.

GPT: Causal Language Modelling

from transformers import GPT2LMHeadModel, GPT2Tokenizer
import torch

tokeniser = GPT2Tokenizer.from_pretrained('gpt2')
model = GPT2LMHeadModel.from_pretrained('gpt2')
model.eval()

prompt = "Machine learning is the study of"
inputs = tokeniser.encode(prompt, return_tensors='pt')

with torch.no_grad():
    # Generate 30 new tokens
    output = model.generate(
        inputs,
        max_new_tokens=30,
        do_sample=True,
        temperature=0.8,          # higher = more random
        top_p=0.9,                # nucleus sampling
        repetition_penalty=1.2,
        pad_token_id=tokeniser.eos_token_id
    )

generated = tokeniser.decode(output[0], skip_special_tokens=True)
print(generated)

BERT for Classification (Fine-tuning)

from transformers import BertForSequenceClassification, BertTokenizer
from torch.optim import AdamW
import torch

model = BertForSequenceClassification.from_pretrained(
    'bert-base-uncased', num_labels=2)
tokeniser = BertTokenizer.from_pretrained('bert-base-uncased')

texts  = ["I loved this movie!", "Terrible waste of time."]
labels = torch.tensor([1, 0])

inputs = tokeniser(texts, padding=True, truncation=True,
                   max_length=128, return_tensors='pt')

optimiser = AdamW(model.parameters(), lr=2e-5)
for step in range(3):
    optimiser.zero_grad()
    outputs = model(**inputs, labels=labels)
    outputs.loss.backward()
    torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
    optimiser.step()
    print(f"Step {step+1}: loss={outputs.loss.item():.4f}")
