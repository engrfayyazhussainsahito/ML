Lesson 6.1: K-Means Clustering

The K-Means Algorithm

K-means partitions n data points into k clusters by minimising the within-cluster sum of squared distances (inertia). It alternates between two steps: (1) assign each point to its nearest centroid, (2) update each centroid to the mean of its assigned points. This is guaranteed to converge but not to find the global optimum.

import numpy as np

def kmeans(X, k, max_iter=300, tol=1e-4, random_state=42):
    rng = np.random.RandomState(random_state)
    # Initialise centroids by sampling k random points
    idx = rng.choice(len(X), k, replace=False)
    centroids = X[idx].copy()

    for iteration in range(max_iter):
        # E-step: assign each point to nearest centroid
        dists = np.linalg.norm(X[:, None] - centroids[None], axis=2)  # (n, k)
        labels = dists.argmin(axis=1)

        # M-step: update centroids
        new_centroids = np.array([X[labels==j].mean(axis=0)
                                  if (labels==j).sum() > 0 else centroids[j]
                                  for j in range(k)])

        shift = np.linalg.norm(new_centroids - centroids)
        centroids = new_centroids
        if shift < tol:
            print(f"Converged at iteration {iteration+1}")
            break

    inertia = sum(np.linalg.norm(X[labels==j] - centroids[j])**2
                  for j in range(k))
    return labels, centroids, inertia

K-Means++ Initialisation

Random initialisation can produce poor results. K-means++ (Arthur & Vassilvitskii, 2007) spreads initial centroids apart using a probability proportional to distance — the first centroid is random, subsequent ones are chosen with probability proportional to their distance² from the nearest existing centroid. This gives a 16-competitive approximation guarantee.

from sklearn.cluster import KMeans
from sklearn.datasets import make_blobs
from sklearn.preprocessing import StandardScaler
import numpy as np

X, y_true = make_blobs(n_samples=500, centers=5, cluster_std=0.8, random_state=42)
X = StandardScaler().fit_transform(X)

# init='k-means++' is the default — always use it
km = KMeans(n_clusters=5, init='k-means++', n_init=10,
            max_iter=300, random_state=42)
km.fit(X)

print(f"Inertia:         {km.inertia_:.2f}")
print(f"Iterations:      {km.n_iter_}")

# Adjusted Rand Index: measures agreement with true labels
from sklearn.metrics import adjusted_rand_score, silhouette_score
ari = adjusted_rand_score(y_true, km.labels_)
sil = silhouette_score(X, km.labels_)
print(f"Adj. Rand Index: {ari:.4f}  (1=perfect, 0=random)")
print(f"Silhouette:      {sil:.4f}  (1=dense, -1=wrong cluster)")

Choosing k: The Elbow Method

import numpy as np
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

inertias, silhouettes = [], []
k_range = range(2, 12)

for k in k_range:
    km = KMeans(n_clusters=k, n_init=10, random_state=42)
    km.fit(X)
    inertias.append(km.inertia_)
    silhouettes.append(silhouette_score(X, km.labels_))

# Elbow: look for the "kink" in the inertia curve
print("k  | Inertia   | Silhouette")
for k, iner, sil in zip(k_range, inertias, silhouettes):
    print(f"{k:2d} | {iner:9.2f} | {sil:.4f}")

best_k_sil = list(k_range)[np.argmax(silhouettes)]
print(f"\nBest k by silhouette: {best_k_sil}")
