Lesson 11.4: Sequence-to-Sequence Models

Sequence-to-Sequence Architecture

Many tasks require mapping a variable-length input sequence to a variable-length output sequence — machine translation, summarisation, question answering, code generation. The seq2seq framework (Sutskever et al., 2014) uses an encoder-decoder architecture: the encoder compresses the source into a context vector, and the decoder generates the target one token at a time conditioned on that context.

Encoder-Decoder with PyTorch

import torch, torch.nn as nn, torch.nn.functional as F

class Encoder(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, n_layers, dropout):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.lstm = nn.LSTM(embed_dim, hidden_dim, n_layers,
                            batch_first=True, dropout=dropout)
        self.dropout = nn.Dropout(dropout)

    def forward(self, src):
        embedded = self.dropout(self.embedding(src))
        outputs, (h, c) = self.lstm(embedded)
        return outputs, h, c   # outputs for attention; h,c as decoder init

class Decoder(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, n_layers, dropout):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.lstm = nn.LSTM(embed_dim, hidden_dim, n_layers,
                            batch_first=True, dropout=dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)
        self.dropout = nn.Dropout(dropout)

    def forward(self, tgt_token, h, c):
        """Decode one token at a time."""
        embedded = self.dropout(self.embedding(tgt_token.unsqueeze(1)))  # (B,1,E)
        out, (h, c) = self.lstm(embedded, (h, c))
        pred = self.fc(out.squeeze(1))   # (B, vocab_size)
        return pred, h, c

class Seq2Seq(nn.Module):
    def __init__(self, encoder, decoder, device):
        super().__init__()
        self.encoder = encoder; self.decoder = decoder; self.device = device

    def forward(self, src, tgt, teacher_forcing_ratio=0.5):
        B, T = tgt.shape
        vocab_size = self.decoder.fc.out_features
        outputs = torch.zeros(B, T, vocab_size).to(self.device)

        _, h, c = self.encoder(src)
        token = tgt[:, 0]   # start with 
        for t in range(1, T):
            pred, h, c = self.decoder(token, h, c)
            outputs[:, t] = pred
            # Teacher forcing: use ground truth or model prediction
            use_teacher = torch.rand(1).item() < teacher_forcing_ratio
            token = tgt[:, t] if use_teacher else pred.argmax(1)
        return outputs

Beam Search Decoding

import torch, heapq

def beam_search(decoder, h, c, sos_id, eos_id, max_len=50, beam_width=5):
    """
    Greedy search picks the best token at each step.
    Beam search maintains `beam_width` candidate sequences simultaneously.
    """
    # (score, token_sequence, h, c)
    beams = [(0.0, [sos_id], h, c)]
    completed = []

    for _ in range(max_len):
        new_beams = []
        for score, tokens, h, c in beams:
            if tokens[-1] == eos_id:
                completed.append((score, tokens)); continue

            token = torch.tensor([tokens[-1]])
            pred, h_new, c_new = decoder(token, h, c)
            log_probs = torch.log_softmax(pred.squeeze(), dim=-1)
            top_k = log_probs.topk(beam_width)

            for log_p, tok in zip(top_k.values, top_k.indices):
                new_score = score - log_p.item()   # min-heap: negate log prob
                new_beams.append((new_score, tokens + [tok.item()], h_new, c_new))

        beams = sorted(new_beams, key=lambda x: x[0])[:beam_width]
        if not beams: break

    all_candidates = completed + [(s, t, None, None) for s, t, _, _ in beams]
    best = min(all_candidates, key=lambda x: x[0] / len(x[1]))
    return best[1]
