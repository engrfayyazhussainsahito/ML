Lesson 16.2: Building an ML Portfolio

Why a Portfolio?

In ML hiring, a strong portfolio demonstrably outweighs academic credentials. A well-presented GitHub repository showing a complete, reproducible project — with clean code, clear documentation, and honest evaluation — signals competence more reliably than a degree. This lesson covers what makes a portfolio compelling and how to build one systematically.

Writing About Results Honestly

import numpy as np
from sklearn.utils import resample

def honest_results_table(model_name, y_test, y_pred, y_prob=None):
    """
    Report results the way a good researcher would.
    """
    from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
    
    acc = accuracy_score(y_test, y_pred)
    f1  = f1_score(y_test, y_pred, average='weighted')
    
    # Bootstrap confidence intervals
    n_bootstrap = 1000
    acc_boot, f1_boot = [], []
    for _ in range(n_bootstrap):
        idx = resample(range(len(y_test)), random_state=None)
        acc_boot.append(accuracy_score(y_test[idx], y_pred[idx]))
        f1_boot.append(f1_score(y_test[idx], y_pred[idx], average='weighted'))
    
    acc_ci = np.percentile(acc_boot, [2.5, 97.5])
    f1_ci  = np.percentile(f1_boot, [2.5, 97.5])
    
    print(f"\n{model_name} Results")
    print(f"  Accuracy: {acc:.3f} (95% CI: [{acc_ci[0]:.3f}, {acc_ci[1]:.3f}])")
    print(f"  F1:       {f1:.3f}  (95% CI: [{f1_ci[0]:.3f}, {f1_ci[1]:.3f}])")
    
    if y_prob is not None:
        auc = roc_auc_score(y_test, y_prob, multi_class='ovr')
        print(f"  AUC:      {auc:.3f}")
    
    # Be honest about limitations
    print("\n  Note: evaluated on held-out test set (n={})".format(len(y_test)))
    print("  Comparison with published baselines where available.")
