Lesson 15.2: Graph Neural Networks

Graph-Structured Data

Many real-world problems involve entities with relationships: molecules (atoms + bonds), social networks (people + friendships), knowledge graphs, citation networks, road networks. Standard neural networks assume fixed-size inputs on a grid — they cannot handle arbitrary graph structure. Graph Neural Networks (GNNs) are designed for exactly this.

Message Passing Framework

import torch, torch.nn as nn

# Graph: nodes with feature vectors, edges defining connectivity
# Notation: N nodes, each with d-dim feature; adjacency A in {0,1}^{N x N}

def gcn_layer(H, A_hat):
    """
    Graph Convolutional Network layer (Kipf and Welling, 2017).
    H:     (N, d) node feature matrix
    A_hat: (N, N) normalised adjacency: D^{-1/2} (A + I) D^{-1/2}
    Returns: (N, d') updated node features
    """
    # Each node aggregates neighbour features + self (via A+I)
    # then applies a linear transform
    return torch.relu(A_hat @ H @ W)   # W: (d, d') learnable

# Normalised adjacency construction
def normalise_adjacency(A):
    """Add self-loops and normalise."""
    N = A.shape[0]
    A_tilde = A + torch.eye(N)   # add self-loops
    D = A_tilde.sum(dim=1)       # degree vector
    D_inv_sqrt = torch.diag(D ** -0.5)
    return D_inv_sqrt @ A_tilde @ D_inv_sqrt

GCN with PyTorch Geometric

import torch, torch.nn.functional as F
from torch_geometric.nn import GCNConv, GATConv, global_mean_pool
from torch_geometric.datasets import Planetoid
from torch_geometric.data import Data

# Node classification on Cora citation network
dataset = Planetoid(root='/tmp/cora', name='Cora')
data = dataset[0]
print(f"Nodes: {data.num_nodes}, Edges: {data.num_edges}")
print(f"Features per node: {data.num_node_features}, Classes: {dataset.num_classes}")

class GCN(torch.nn.Module):
    def __init__(self, in_channels, hidden, out_channels):
        super().__init__()
        self.conv1 = GCNConv(in_channels, hidden)
        self.conv2 = GCNConv(hidden, out_channels)

    def forward(self, x, edge_index):
        x = F.relu(self.conv1(x, edge_index))
        x = F.dropout(x, p=0.5, training=self.training)
        return self.conv2(x, edge_index)

model = GCN(dataset.num_node_features, 64, dataset.num_classes)
opt   = torch.optim.Adam(model.parameters(), lr=0.01, weight_decay=5e-4)

for epoch in range(200):
    model.train(); opt.zero_grad()
    out  = model(data.x, data.edge_index)
    loss = F.cross_entropy(out[data.train_mask], data.y[data.train_mask])
    loss.backward(); opt.step()

model.eval()
with torch.no_grad():
    pred = model(data.x, data.edge_index).argmax(dim=1)
    acc  = (pred[data.test_mask] == data.y[data.test_mask]).float().mean()
print(f"Test accuracy: {acc:.4f}")   # typically ~81% on Cora

Graph Attention Networks (GAT)

from torch_geometric.nn import GATConv

class GAT(torch.nn.Module):
    """GAT: learn attention weights over neighbours instead of fixed normalisation."""
    def __init__(self, in_ch, hidden, out_ch, heads=8):
        super().__init__()
        self.conv1 = GATConv(in_ch, hidden, heads=heads, dropout=0.6)
        self.conv2 = GATConv(hidden*heads, out_ch, heads=1, concat=False, dropout=0.6)

    def forward(self, x, edge_index):
        x = F.elu(self.conv1(x, edge_index))
        return self.conv2(x, edge_index)

gat = GAT(dataset.num_node_features, 8, dataset.num_classes)
print(f"GAT parameters: {sum(p.numel() for p in gat.parameters()):,}")
