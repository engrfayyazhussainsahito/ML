Lesson 8.1: End-to-End Project

The Complete ML Project

This lesson walks through every stage of an end-to-end ML project on the Titanic dataset — the canonical learning example. Every line of code reflects real practices used in production ML: defensive data loading, systematic EDA, leakage-free preprocessing, model comparison, and confident final evaluation.

Stage 1: Load and Inspect

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split

# Load
df = pd.read_csv("titanic.csv")

print(f"Shape: {df.shape}")
print(f"Columns: {df.columns.tolist()}")
print(f"Dtypes:\n{df.dtypes}")
print(f"Missing:\n{df.isnull().mean().round(3)}")
print(f"Target distribution:\n{df['Survived'].value_counts(normalize=True)}")

# Split immediately — all EDA on train only
df_train, df_test = train_test_split(df, test_size=0.2, 
                                     stratify=df['Survived'], random_state=42)
print(f"Train: {len(df_train)}, Test: {len(df_test)}")

Stage 2: EDA on Train Only

import pandas as pd, numpy as np

# Survival rate by feature
print("By sex:")
print(df_train.groupby('Sex')['Survived'].mean().round(3))

print("\nBy class:")
print(df_train.groupby('Pclass')['Survived'].mean().round(3))

print("\nAge statistics:")
print(df_train['Age'].describe().round(2))

# Correlations with target
num_features = df_train.select_dtypes(include=np.number).columns
corr_with_target = df_train[num_features].corrwith(df_train['Survived'])
print("\nCorrelations with Survived:")
print(corr_with_target.sort_values(key=abs, ascending=False))

Stage 3: Feature Engineering and Pipeline

from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.impute import SimpleImputer

def engineer_features(df):
    """Create new features from domain knowledge."""
    df = df.copy()
    df['FamilySize'] = df['SibSp'] + df['Parch'] + 1
    df['IsAlone'] = (df['FamilySize'] == 1).astype(int)
    df['Title'] = df['Name'].str.extract(r' ([A-Za-z]+)\.', expand=False)
    df['Title'] = df['Title'].replace(
        ['Lady','Countess','Capt','Col','Don','Dr','Major','Rev','Sir','Jonkheer','Dona'], 'Rare')
    df['Title'] = df['Title'].replace({'Mlle':'Miss','Ms':'Miss','Mme':'Mrs'})
    return df

df_train_eng = engineer_features(df_train)

num_cols = ['Age', 'Fare', 'FamilySize']
cat_cols = ['Sex', 'Pclass', 'Embarked', 'Title']

preprocessor = ColumnTransformer([
    ('num', Pipeline([('imp', SimpleImputer(strategy='median')),
                      ('sc',  StandardScaler())]), num_cols),
    ('cat', Pipeline([('imp', SimpleImputer(strategy='most_frequent')),
                      ('ohe', OneHotEncoder(handle_unknown='ignore',
                                            sparse_output=False))]), cat_cols),
])

Stage 4: Model Comparison

from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.model_selection import StratifiedKFold, cross_validate
import numpy as np

X_train = df_train_eng[num_cols + cat_cols]
y_train = df_train_eng['Survived']

models = {
    'LogReg': LogisticRegression(C=0.1, max_iter=1000),
    'RF':     RandomForestClassifier(n_estimators=200, max_depth=6, random_state=42),
    'GBM':    GradientBoostingClassifier(n_estimators=300, max_depth=3,
                                          learning_rate=0.05, random_state=42),
}

cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
print(f"{'Model':10s}  {'Accuracy':>10s}  {'AUC':>10s}")
for name, clf in models.items():
    pipe = Pipeline([('pre', preprocessor), ('clf', clf)])
    res = cross_validate(pipe, X_train, y_train, cv=cv,
                         scoring=['accuracy', 'roc_auc'], n_jobs=-1)
    acc = res['test_accuracy'].mean()
    auc = res['test_roc_auc'].mean()
    print(f"{name:10s}  {acc:.4f}       {auc:.4f}")

Stage 5: Final Evaluation

from sklearn.metrics import classification_report, roc_auc_score
import numpy as np

# Refit best model on all training data
best_pipe = Pipeline([('pre', preprocessor),
                      ('clf', GradientBoostingClassifier(n_estimators=300,
                                max_depth=3, learning_rate=0.05, random_state=42))])
best_pipe.fit(X_train, y_train)

# Evaluate on held-out test set (done ONCE)
df_test_eng = engineer_features(df_test)
X_test = df_test_eng[num_cols + cat_cols]
y_test = df_test_eng['Survived']

y_pred = best_pipe.predict(X_test)
y_prob = best_pipe.predict_proba(X_test)[:, 1]

print(classification_report(y_test, y_pred, target_names=['Did not survive','Survived']))
print(f"AUC: {roc_auc_score(y_test, y_prob):.4f}")
