Lesson 10.2: CNN Architectures

The Architecture Family Tree

CNN architectures have evolved dramatically since 2012. Each generation introduced an architectural innovation that addressed the limitations of the previous one — from AlexNet's depth and ReLU through VGG's simplicity, GoogLeNet's efficiency, ResNet's residual connections, to EfficientNet's compound scaling.

graph LR
  A[LeNet 1998] -->|depth+ReLU| B[AlexNet 2012]
  B -->|uniform 3x3| C[VGG 2014]
  B -->|inception module| D[GoogLeNet 2014]
  C -->|skip connections| E[ResNet 2015]
  E -->|dense connections| F[DenseNet 2017]
  E -->|compound scaling| G[EfficientNet 2019]

ResNet: Residual Connections

He et al. (2015) observed that deeper networks were performing worse than shallower ones — not due to overfitting but because gradient vanishing prevented training. The residual connection y = F(x) + x adds the input to the output, creating a gradient highway that bypasses the non-linear transformation. ResNet-152 was trained with 152 layers — impossible without skip connections.

import torch, torch.nn as nn

class ResidualBlock(nn.Module):
    """Standard ResNet residual block."""
    def __init__(self, in_channels, out_channels, stride=1):
        super().__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, 3,
                               stride=stride, padding=1, bias=False)
        self.bn1   = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, 3,
                               padding=1, bias=False)
        self.bn2   = nn.BatchNorm2d(out_channels)
        self.relu  = nn.ReLU(inplace=True)

        # 1x1 conv to match dimensions when stride>1 or channels change
        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, 1, stride=stride, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += self.shortcut(x)     # skip connection
        return self.relu(out)

# Build a small ResNet-18 style network
class ResNet(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.stem = nn.Sequential(
            nn.Conv2d(3, 64, 7, stride=2, padding=3, bias=False),
            nn.BatchNorm2d(64), nn.ReLU(inplace=True),
            nn.MaxPool2d(3, stride=2, padding=1)
        )
        self.layer1 = self._make_layer(64,  64,  2, stride=1)
        self.layer2 = self._make_layer(64,  128, 2, stride=2)
        self.layer3 = self._make_layer(128, 256, 2, stride=2)
        self.layer4 = self._make_layer(256, 512, 2, stride=2)
        self.head   = nn.Sequential(nn.AdaptiveAvgPool2d(1), nn.Flatten(),
                                    nn.Linear(512, num_classes))

    def _make_layer(self, in_ch, out_ch, n_blocks, stride):
        layers = [ResidualBlock(in_ch, out_ch, stride)]
        for _ in range(n_blocks - 1):
            layers.append(ResidualBlock(out_ch, out_ch))
        return nn.Sequential(*layers)

    def forward(self, x):
        return self.head(self.layer4(self.layer3(self.layer2(self.layer1(self.stem(x))))))

model = ResNet(num_classes=10)
x = torch.randn(4, 3, 224, 224)
print(model(x).shape)   # (4, 10)
print(f"Parameters: {sum(p.numel() for p in model.parameters()):,}")

Using Pretrained Models from torchvision

import torchvision.models as models
import torch

# Load ImageNet-pretrained ResNet-50
resnet = models.resnet50(weights=models.ResNet50_Weights.IMAGENET1K_V2)
print(f"ResNet-50 parameters: {sum(p.numel() for p in resnet.parameters()):,}")

# EfficientNet: SOTA accuracy/efficiency tradeoff
effnet = models.efficientnet_b0(weights=models.EfficientNet_B0_Weights.IMAGENET1K_V1)

# Vision Transformer (ViT): attention-based image model
vit = models.vit_b_16(weights=models.ViT_B_16_Weights.IMAGENET1K_V1)

# Inference
model.eval()
with torch.no_grad():
    x = torch.randn(1, 3, 224, 224)
    logits = resnet(x)   # (1, 1000) — 1000 ImageNet classes
    probs  = torch.softmax(logits, dim=1)
    top5   = probs.topk(5)
    print(f"Top-5 class probabilities: {top5.values.squeeze().numpy().round(4)}")
