Lesson 13.1: Autoencoders and VAEs

Autoencoders

An autoencoder learns to compress data into a low-dimensional latent representation and reconstruct the original from it. The encoder E maps input x to latent code z; the decoder D maps z back to x̂. Training minimises reconstruction error: L = ||x − D(E(x))||². The bottleneck forces the network to learn a compact, informative representation.

import torch, torch.nn as nn

class Autoencoder(nn.Module):
    def __init__(self, input_dim=784, latent_dim=32):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, 512), nn.ReLU(),
            nn.Linear(512, 128),       nn.ReLU(),
            nn.Linear(128, latent_dim)
        )
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 128), nn.ReLU(),
            nn.Linear(128, 512),        nn.ReLU(),
            nn.Linear(512, input_dim),  nn.Sigmoid()
        )

    def forward(self, x):
        z    = self.encoder(x)
        x_hat = self.decoder(z)
        return x_hat, z

ae = Autoencoder(784, 32)
x = torch.randn(64, 784)
x_hat, z = ae(x)
print(f"Input:   {x.shape}")      # (64, 784)
print(f"Latent:  {z.shape}")      # (64, 32)
print(f"Output:  {x_hat.shape}")  # (64, 784)

criterion = nn.MSELoss()
loss = criterion(x_hat, x)
print(f"MSE: {loss.item():.4f}")

The Problem with Standard Autoencoders

Standard autoencoders learn a discrete, irregular latent space. If you sample a random point z from the latent space and decode it, the output is typically meaningless — the encoder only populates isolated regions for each training example. We cannot use the latent space for generation.

Variational Autoencoders (VAE)

VAEs (Kingma and Welling, 2013) solve this by making the encoder output a probability distribution q(z|x) = N(μ, σ²) instead of a single point. The decoder must reconstruct from samples drawn from this distribution, forcing the latent space to be smooth and continuous. Crucially, sampling is made differentiable via the reparameterisation trick: z = μ + σ·ε, where ε ~ N(0,1).

import torch, torch.nn as nn, torch.nn.functional as F

class VAE(nn.Module):
    def __init__(self, input_dim=784, latent_dim=20):
        super().__init__()
        # Encoder outputs mean and log-variance
        self.fc_enc   = nn.Sequential(nn.Linear(input_dim, 512), nn.ReLU())
        self.fc_mu    = nn.Linear(512, latent_dim)
        self.fc_logvar = nn.Linear(512, latent_dim)

        # Decoder
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 512), nn.ReLU(),
            nn.Linear(512, input_dim),  nn.Sigmoid()
        )

    def encode(self, x):
        h  = self.fc_enc(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)
        return mu, logvar

    def reparameterise(self, mu, logvar):
        """z = mu + sigma * epsilon   (differentiable sampling)"""
        if self.training:
            std = torch.exp(0.5 * logvar)
            eps = torch.randn_like(std)
            return mu + eps * std
        return mu   # deterministic at inference

    def decode(self, z):
        return self.decoder(z)

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterise(mu, logvar)
        return self.decode(z), mu, logvar

def vae_loss(x_hat, x, mu, logvar, beta=1.0):
    """
    ELBO = Reconstruction loss + beta * KL divergence
    KL( q(z|x) || N(0,I) ) = -0.5 * sum(1 + logvar - mu^2 - e^logvar)
    """
    recon = F.binary_cross_entropy(x_hat, x, reduction='sum')
    kld   = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp())
    return (recon + beta * kld) / x.size(0)

vae = VAE(784, 20)
x   = torch.rand(64, 784)
x_hat, mu, logvar = vae(x)
loss = vae_loss(x_hat, x, mu, logvar, beta=1.0)
print(f"VAE ELBO loss: {loss.item():.2f}")

# Generation: sample z ~ N(0,I), decode
z_sample = torch.randn(16, 20)
generated = vae.decode(z_sample)
print(f"Generated: {generated.shape}")   # (16, 784)

Beta-VAE and Disentanglement

# beta > 1 encourages disentangled latent factors
# Each dimension of z should correspond to one interpretable factor
# e.g. for face images: z[0]=pose, z[1]=lighting, z[2]=expression

# beta=4 works well for many datasets
loss = vae_loss(x_hat, x, mu, logvar, beta=4.0)

# Interpolation in latent space
z1 = vae.encode(x[:1])[0]    # latent of image 1
z2 = vae.encode(x[1:2])[0]   # latent of image 2

interpolations = []
for alpha in torch.linspace(0, 1, 8):
    z_interp = (1 - alpha) * z1 + alpha * z2
    interpolations.append(vae.decode(z_interp))

# Smooth interpolation only possible in a well-structured latent space
