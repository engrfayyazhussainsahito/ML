Lesson 11.2: RNNs and the Vanishing Gradient

The Recurrent Cell

An RNN processes a sequence one timestep at a time, maintaining a hidden state hₜ that summarises all past inputs. The same weight matrix W is applied at every timestep — this is parameter sharing across time, analogous to how CNNs share weights across space.

hₜ = tanh(Wₕhₜ₋₁ + Wₓxₜ + b)

The hidden state hₜ is the "memory" of the network. The final hidden state h_T (or all hₜ) is passed to an output layer.

RNN from Scratch

import numpy as np

class SimpleRNN:
    def __init__(self, input_dim, hidden_dim, output_dim, lr=0.01):
        scale = np.sqrt(2 / hidden_dim)
        self.Wh = np.random.randn(hidden_dim, hidden_dim) * scale
        self.Wx = np.random.randn(input_dim,  hidden_dim) * scale
        self.Wy = np.random.randn(hidden_dim, output_dim) * scale
        self.bh = np.zeros(hidden_dim)
        self.by = np.zeros(output_dim)
        self.lr = lr

    def forward(self, X):
        """X: (seq_len, batch, input_dim)"""
        T, B, _ = X.shape
        h = np.zeros((B, self.Wh.shape[0]))
        self.hs = [h]     # cache hidden states for BPTT
        self.xs = []

        for t in range(T):
            xt = X[t]     # (B, input_dim)
            h  = np.tanh(xt @ self.Wx + h @ self.Wh + self.bh)
            self.hs.append(h); self.xs.append(xt)

        y_hat = h @ self.Wy + self.by   # predict from last hidden state
        return y_hat, self.hs

rnn = SimpleRNN(input_dim=5, hidden_dim=32, output_dim=1)
X = np.random.randn(10, 8, 5)   # (seq_len=10, batch=8, features=5)
out, hidden = rnn.forward(X)
print(f"Output shape: {out.shape}")   # (8, 1)

Backpropagation Through Time (BPTT)

import torch, torch.nn as nn

class RNNClassifier(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim, n_layers=2, dropout=0.3):
        super().__init__()
        self.rnn = nn.RNN(
            input_dim, hidden_dim,
            num_layers=n_layers,
            batch_first=True,
            dropout=dropout if n_layers > 1 else 0,
            nonlinearity='tanh'
        )
        self.fc = nn.Linear(hidden_dim, output_dim)

    def forward(self, x, lengths=None):
        # x: (batch, seq_len, input_dim)
        out, h_n = self.rnn(x)    # out: (batch, seq_len, hidden)
        return self.fc(out[:, -1, :])   # use last timestep

model = RNNClassifier(input_dim=10, hidden_dim=128, output_dim=2)
x = torch.randn(32, 20, 10)   # (batch=32, seq_len=20, features=10)
logits = model(x)
print(logits.shape)   # (32, 2)

The Vanishing Gradient Problem

import torch, torch.nn as nn
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Visualise gradient flow through time
rnn = nn.RNN(10, 64, batch_first=True)
x = torch.randn(1, 100, 10, requires_grad=True)   # long sequence
out, _ = rnn(x)
loss = out[0, -1, :].sum()   # use only last timestep
loss.backward()

# Gradient w.r.t. x at each timestep
grad_norms = x.grad[0].norm(dim=1).detach().numpy()  # (100,)
print("Gradient norms at early timesteps (should be near 0):")
print(grad_norms[:10].round(6))
print("Gradient norms at late timesteps:")
print(grad_norms[-10:].round(6))

# Gradient clipping: prevents exploding gradients
optimiser = torch.optim.SGD(rnn.parameters(), lr=0.01)
for _ in range(10):
    optimiser.zero_grad()
    out, _ = rnn(torch.randn(32, 50, 10))
    out[:,-1,:].sum().backward()
    nn.utils.clip_grad_norm_(rnn.parameters(), max_norm=1.0)   # clip!
    optimiser.step()
