Lesson 11.1: Sequences and Time Series

What Makes Sequential Data Special?

Sequential data has an inherent order — the meaning of an observation depends on what came before. A sentence's next word depends on prior words; tomorrow's stock price depends on prior prices; a protein's function depends on its amino acid sequence. Standard feedforward networks treat each input independently and cannot exploit this temporal dependency.

Representing Sequences

import numpy as np

# A time series as a 2D array: (timesteps, features)
T, n_features = 100, 5
x_ts = np.random.randn(T, n_features)

# Windowing: create supervised examples from a time series
def create_windows(x, lookback, horizon=1):
    """
    x: (T, features)
    Returns X: (n_windows, lookback, features), y: (n_windows, features)
    """
    X, y = [], []
    for t in range(lookback, len(x) - horizon + 1):
        X.append(x[t-lookback:t])
        y.append(x[t:t+horizon])
    return np.array(X), np.array(y)

X_windows, y_windows = create_windows(x_ts, lookback=10, horizon=1)
print(f"X shape: {X_windows.shape}")   # (89, 10, 5)
print(f"y shape: {y_windows.shape}")   # (89, 1, 5)

# For text: tokenise into integer IDs
sentence = "the cat sat on the mat"
vocab = {w: i for i, w in enumerate(set(sentence.split()))}
token_ids = [vocab[w] for w in sentence.split()]
print(f"Token IDs: {token_ids}")

Stationarity and Preprocessing

import numpy as np
from scipy import stats

# Augmented Dickey-Fuller test for stationarity
# H0: series has a unit root (non-stationary)
def check_stationarity(series, name=""):
    from statsmodels.tsa.stattools import adfuller
    result = adfuller(series)
    p_value = result[1]
    stationary = p_value < 0.05
    print(f"{name}: ADF p={p_value:.4f} -> {'stationary' if stationary else 'NON-STATIONARY'}")
    return stationary

np.random.seed(42)
# Non-stationary: random walk
rw = np.cumsum(np.random.randn(500))
check_stationarity(rw, "Random walk")

# Differencing to achieve stationarity
rw_diff = np.diff(rw)
check_stationarity(rw_diff, "Differenced random walk")

# Log-return for financial series
prices = np.abs(np.cumsum(np.random.randn(500)) + 100)
log_returns = np.log(prices[1:] / prices[:-1])
check_stationarity(log_returns, "Log returns")

PyTorch Sequence Batching

import torch
from torch.nn.utils.rnn import pad_sequence, pack_padded_sequence

# Variable-length sequences: pad to the same length
sequences = [torch.tensor([1,2,3,4]), torch.tensor([5,6]), torch.tensor([7,8,9])]
lengths   = torch.tensor([len(s) for s in sequences])

padded = pad_sequence(sequences, batch_first=True, padding_value=0)
print(f"Padded: {padded}")   # (3, 4) — padded with zeros
print(f"Lengths: {lengths}")

# PackedSequence: efficient computation ignoring padding
from torch.nn.utils.rnn import pack_padded_sequence, pad_packed_sequence
packed = pack_padded_sequence(padded, lengths, batch_first=True, enforce_sorted=True)
# RNN processes only the non-padded elements — 2x speedup on short sequences
