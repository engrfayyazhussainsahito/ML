Lesson 7.1: Feature Scaling and Encoding

Feature Scaling

Distance-based models (KNN, SVM, neural networks, PCA) are sensitive to feature scale. A feature measured in thousands will dominate Euclidean distance over a feature measured in [0,1]. Tree-based models (random forests, XGBoost) are invariant to monotonic transformations and do not require scaling.

from sklearn.preprocessing import StandardScaler, MinMaxScaler, RobustScaler
import numpy as np

X = np.array([[100, 0.01, 5000],
              [200, 0.05, 3000],
              [150, 0.02, 8000],
              [300, 0.10, 2000]])

# StandardScaler: z-score normalisation -> mean=0, std=1
# Best for: normally distributed features, PCA, neural networks
std = StandardScaler()
print("StandardScaler:")
print(std.fit_transform(X).round(3))

# MinMaxScaler: scale to [0, 1]
# Best for: neural networks, when you need bounded output
mm = MinMaxScaler()
print("\nMinMaxScaler:")
print(mm.fit_transform(X).round(3))

# RobustScaler: uses median and IQR instead of mean and std
# Best for: datasets with outliers
rb = RobustScaler()
print("\nRobustScaler:")
print(rb.fit_transform(X).round(3))

Categorical Encoding

import pandas as pd
import numpy as np
from sklearn.preprocessing import OneHotEncoder, OrdinalEncoder

df = pd.DataFrame({
    "colour": ["red", "blue", "green", "red", "blue"],
    "size":   ["S", "M", "L", "XL", "M"],
    "price":  [10, 20, 15, 25, 18],
    "sold":   [1, 0, 1, 1, 0]
})

# --- One-Hot Encoding (nominal: no order) ---
ohe = OneHotEncoder(sparse_output=False, drop='first')
colour_encoded = ohe.fit_transform(df[["colour"]])
print("One-hot (colour):", ohe.get_feature_names_out())

# --- Ordinal Encoding (ordered categories) ---
size_order = [["S", "M", "L", "XL"]]
oe = OrdinalEncoder(categories=size_order)
df["size_enc"] = oe.fit_transform(df[["size"]])
print("Ordinal (size):", df["size_enc"].values)

# --- Target Encoding (high-cardinality categoricals) ---
# Replace each category with its mean target value
# MUST be done with cross-validation to avoid leakage
from sklearn.model_selection import KFold
import numpy as np

def target_encode_cv(X_col, y, n_folds=5):
    encoded = np.zeros(len(X_col))
    kf = KFold(n_splits=n_folds, shuffle=True, random_state=42)
    for tr_idx, val_idx in kf.split(X_col):
        mapping = y[tr_idx].groupby(X_col[tr_idx]).mean()
        encoded[val_idx] = X_col[val_idx].map(mapping).fillna(y[tr_idx].mean())
    return encoded

Transforming Skewed Features

import numpy as np
from sklearn.preprocessing import PowerTransformer

# Log transform: compresses right-skewed distributions
income = np.array([25000, 30000, 45000, 60000, 500000, 1000000])
log_income = np.log1p(income)   # log1p(x) = log(1+x), handles zero safely

# Box-Cox / Yeo-Johnson (handles negatives) via sklearn
pt = PowerTransformer(method='yeo-johnson')
income_transformed = pt.fit_transform(income.reshape(-1,1))

from scipy import stats
# Check normality before and after
stat_orig, p_orig = stats.normaltest(income)
stat_tr, p_tr     = stats.normaltest(income_transformed.ravel())
print(f"Original normality p-value:    {p_orig:.4f}")
print(f"Transformed normality p-value: {p_tr:.4f}")
# Higher p-value = more Gaussian
