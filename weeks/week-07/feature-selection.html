Lesson 7.2: Feature Selection

Why Feature Selection?

Adding irrelevant features increases noise, slows training, and can degrade performance (the curse of dimensionality). Feature selection reduces the feature matrix to the most informative subset, improving generalisation, interpretability, and computational efficiency.

Filter Methods

from sklearn.feature_selection import (
    VarianceThreshold, SelectKBest, f_classif, mutual_info_classif
)
from sklearn.datasets import make_classification
import numpy as np

X, y = make_classification(n_samples=1000, n_features=50,
                            n_informative=15, random_state=42)

# Remove constant/quasi-constant features
vt = VarianceThreshold(threshold=0.01)
X_var = vt.fit_transform(X)
print(f"After variance filter: {X_var.shape[1]} features")

# Univariate: select top k features by ANOVA F-statistic
sel_f = SelectKBest(f_classif, k=20)
X_kbest_f = sel_f.fit_transform(X, y)
print(f"ANOVA F: selected {X_kbest_f.shape[1]} features")

# Mutual information: captures non-linear relationships
sel_mi = SelectKBest(mutual_info_classif, k=20)
X_kbest_mi = sel_mi.fit_transform(X, y)
scores_mi = sel_mi.scores_
top_features = np.argsort(scores_mi)[::-1][:10]
print(f"Top 10 by MI: {top_features}")

Wrapper Methods: Recursive Feature Elimination

from sklearn.feature_selection import RFE, RFECV
from sklearn.ensemble import RandomForestClassifier

# RFE: train model, remove least important feature, repeat
rfe = RFE(
    estimator=RandomForestClassifier(n_estimators=100, random_state=42),
    n_features_to_select=15,
    step=0.1,   # remove 10% of remaining features per round
    verbose=0
)
rfe.fit(X, y)
print(f"Selected features: {rfe.support_.sum()}")
print(f"Feature rankings (1=selected): {rfe.ranking_[:10]}")

# RFECV: find optimal number of features with CV
rfecv = RFECV(
    estimator=RandomForestClassifier(n_estimators=50, random_state=42),
    step=1, cv=5, scoring='accuracy', n_jobs=-1, min_features_to_select=5
)
rfecv.fit(X, y)
print(f"Optimal features: {rfecv.n_features_}")
print(f"CV score: {rfecv.cv_results_['mean_test_score'].max():.4f}")

Embedded Methods: SHAP

import shap
from sklearn.ensemble import GradientBoostingClassifier
import numpy as np

model = GradientBoostingClassifier(n_estimators=200, random_state=42)
model.fit(X, y)

explainer = shap.TreeExplainer(model)
shap_values = explainer.shap_values(X[:200])   # compute for subset

# Mean absolute SHAP value = global feature importance
mean_shap = np.abs(shap_values).mean(axis=0)
ranking = np.argsort(mean_shap)[::-1]
print("Top 10 features by SHAP:")
for i in range(10):
    print(f"  Feature {ranking[i]:2d}: {mean_shap[ranking[i]]:.4f}")

# Select features with SHAP > threshold
threshold = 0.01
selected = np.where(mean_shap > threshold)[0]
print(f"Selected {len(selected)} features with SHAP > {threshold}")
