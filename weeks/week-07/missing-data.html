Lesson 7.3: Handling Missing Data

Types of Missingness

Understanding why data is missing is as important as knowing how much. The mechanism determines which imputation strategies are valid:

MechanismDefinitionExampleSafe to impute?
MCARMissing completely at random — independent of all variablesRandom sensor failureYes
MARMissing at random — depends on observed variables, not missing onesMen less likely to report weightYes (with care)
MNARMissing not at random — depends on the missing value itselfHigh earners skip income fieldRequires domain knowledge

Diagnosing Missing Data

import pandas as pd
import numpy as np

df = pd.read_csv("data.csv")

# Summary of missingness
missing_summary = pd.DataFrame({
    'count':   df.isnull().sum(),
    'pct':     df.isnull().mean() * 100,
    'dtype':   df.dtypes
}).query('count > 0').sort_values('pct', ascending=False)
print(missing_summary)

# Test for MCAR: is missingness correlated with other features?
for col in df.columns[df.isnull().any()]:
    missing_mask = df[col].isnull().astype(int)
    for other in df.select_dtypes(include=np.number).columns:
        if other != col and df[other].isnull().sum() == 0:
            corr = missing_mask.corr(df[other])
            if abs(corr) > 0.1:
                print(f"Missingness in {col} correlated with {other}: r={corr:.3f}")

Imputation Strategies

from sklearn.impute import SimpleImputer, KNNImputer
from sklearn.experimental import enable_iterative_imputer
from sklearn.impute import IterativeImputer
import numpy as np

X = np.random.randn(100, 5)
# Introduce ~10% missingness at random
X_missing = X.copy()
X_missing[np.random.rand(*X.shape) < 0.1] = np.nan

# 1. Simple imputation: mean/median/mode/constant
for strategy in ['mean', 'median']:
    imp = SimpleImputer(strategy=strategy)
    X_imp = imp.fit_transform(X_missing)
    mse = np.nanmean((X_imp - X)**2)
    print(f"{strategy:8s} MSE: {mse:.4f}")

# 2. KNN imputation: impute from k nearest non-missing neighbours
knn_imp = KNNImputer(n_neighbors=5)
X_knn = knn_imp.fit_transform(X_missing)
mse_knn = np.nanmean((X_knn - X)**2)
print(f"KNN      MSE: {mse_knn:.4f}")

# 3. Iterative (MICE): model each feature as function of others
mice = IterativeImputer(max_iter=10, random_state=42)
X_mice = mice.fit_transform(X_missing)
mse_mice = np.nanmean((X_mice - X)**2)
print(f"MICE     MSE: {mse_mice:.4f}")

Missing Indicator Features

from sklearn.impute import SimpleImputer, MissingIndicator
from sklearn.pipeline import Pipeline, FeatureUnion
import numpy as np

# Adding a binary "was_missing" column preserves the missingness signal
# This is valuable when missingness itself is informative (MNAR)

indicator = MissingIndicator(features='missing-only')
imputer   = SimpleImputer(strategy='mean')

# FeatureUnion: append indicator columns to imputed columns
combined = FeatureUnion([
    ('imputed',   imputer),
    ('indicator', indicator)
])

X_missing = np.array([[1.0, np.nan], [2.0, 3.0], [np.nan, 4.0], [5.0, 6.0]])
X_combined = combined.fit_transform(X_missing)
print("Shape:", X_combined.shape)   # (4, 4): 2 imputed + 2 indicators
print(X_combined)
