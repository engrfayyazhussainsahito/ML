Lesson 5.3: Stacking and Blending

Stacking: Learning to Combine

Stacking (stacked generalisation) trains a meta-learner to optimally combine the predictions of several base models. Unlike simple averaging, the meta-learner can learn when to trust each base model — for example, weighting a gradient booster higher on structured features and a neural network higher on embeddings.

Stacking Implementation

from sklearn.ensemble import StackingClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.svm import SVC
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split, cross_val_score
import numpy as np

X, y = make_classification(n_samples=2000, n_features=20, random_state=42)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)

base_learners = [
    ('rf',  RandomForestClassifier(n_estimators=200, random_state=42)),
    ('gbm', GradientBoostingClassifier(n_estimators=200, random_state=42)),
    ('svc', SVC(kernel='rbf', probability=True, random_state=42)),
]

# Meta-learner: logistic regression on out-of-fold base model predictions
stack = StackingClassifier(
    estimators=base_learners,
    final_estimator=LogisticRegression(),
    cv=5,                         # out-of-fold strategy avoids leakage
    stack_method='predict_proba', # pass probabilities to meta-learner
    n_jobs=-1
)

stack.fit(X_train, y_train)
print(f"Stack accuracy: {stack.score(X_test, y_test):.4f}")

# Compare individual base learners
for name, clf in base_learners:
    clf.fit(X_train, y_train)
    print(f"{name} accuracy: {clf.score(X_test, y_test):.4f}")

Out-of-Fold Predictions (Manual)

The critical detail: base model predictions for the meta-learner must be out-of-fold — the meta-learner must never see predictions made on data the base model trained on. StackingClassifier handles this automatically; here is the manual version for understanding.

from sklearn.model_selection import StratifiedKFold
import numpy as np

def make_oof_predictions(model, X, y, n_folds=5):
    """Generate out-of-fold predictions."""
    oof = np.zeros((len(X), 2))  # probability for each class
    skf = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=42)

    for fold, (train_idx, val_idx) in enumerate(skf.split(X, y)):
        model.fit(X[train_idx], y[train_idx])
        oof[val_idx] = model.predict_proba(X[val_idx])

    return oof

rf_oof  = make_oof_predictions(RandomForestClassifier(n_estimators=100), X_train, y_train)
gbm_oof = make_oof_predictions(GradientBoostingClassifier(n_estimators=100), X_train, y_train)

# Stack OOF predictions as features for the meta-learner
meta_features_train = np.hstack([rf_oof, gbm_oof])  # (n_train, 4)

meta = LogisticRegression()
meta.fit(meta_features_train, y_train)

# For test: refit base models on full train, then predict
rf_full  = RandomForestClassifier(n_estimators=100).fit(X_train, y_train)
gbm_full = GradientBoostingClassifier(n_estimators=100).fit(X_train, y_train)
meta_test = np.hstack([rf_full.predict_proba(X_test),
                        gbm_full.predict_proba(X_test)])
print(f"Manual stack accuracy: {(meta.predict(meta_test) == y_test).mean():.4f}")

Blending vs Stacking

BlendingStacking
Meta-learner training dataHoldout setOut-of-fold predictions
Data efficiencyLowerHigher
Leakage riskLowerHigher if done wrong
ImplementationSimplerMore complex
Common useKaggle late-stageProduction ensembles
