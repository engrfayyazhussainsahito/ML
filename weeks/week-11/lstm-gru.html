Lesson 11.3: LSTMs and GRUs

The LSTM Solution

Hochreiter and Schmidhuber (1997) designed the Long Short-Term Memory cell to address the vanishing gradient problem. The key innovation: a separate cell state cₜ — a "conveyor belt" of information — that flows through time with only multiplicative interactions (gates), allowing gradients to flow unimpeded across many timesteps.

The Four Gates

import torch, torch.nn as nn

class LSTMCell(nn.Module):
    """Manual LSTM cell for pedagogical clarity."""
    def __init__(self, input_dim, hidden_dim):
        super().__init__()
        # All four gates computed in one matrix multiply for efficiency
        self.linear = nn.Linear(input_dim + hidden_dim, 4 * hidden_dim)

    def forward(self, x, h_prev, c_prev):
        """x: (batch, input_dim); h_prev, c_prev: (batch, hidden_dim)"""
        combined = torch.cat([x, h_prev], dim=1)   # (batch, input+hidden)
        gates = self.linear(combined)               # (batch, 4*hidden)
        input_gate, forget_gate, cell_gate, output_gate = gates.chunk(4, dim=1)

        i = torch.sigmoid(input_gate)    # how much of new info to let in
        f = torch.sigmoid(forget_gate)   # how much of old cell to keep
        g = torch.tanh(cell_gate)        # candidate new cell content
        o = torch.sigmoid(output_gate)   # how much of cell to output

        c = f * c_prev + i * g    # new cell state: update old, add new
        h = o * torch.tanh(c)     # new hidden state: filtered cell

        return h, c

# Using PyTorch's optimised LSTM
lstm = nn.LSTM(input_size=10, hidden_size=128, num_layers=2,
               batch_first=True, dropout=0.2, bidirectional=False)
x = torch.randn(32, 50, 10)   # (batch, seq_len, features)
out, (h_n, c_n) = lstm(x)
print(f"Output:      {out.shape}")   # (32, 50, 128)
print(f"Hidden (h):  {h_n.shape}")  # (2, 32, 128) — 2 layers
print(f"Cell (c):    {c_n.shape}")  # (2, 32, 128)

GRU: Gated Recurrent Unit

import torch, torch.nn as nn

# GRU simplifies LSTM: merges forget/input gates, removes cell state
class GRUCell(nn.Module):
    def __init__(self, input_dim, hidden_dim):
        super().__init__()
        self.reset_gate  = nn.Linear(input_dim + hidden_dim, hidden_dim)
        self.update_gate = nn.Linear(input_dim + hidden_dim, hidden_dim)
        self.new_gate    = nn.Linear(input_dim + hidden_dim, hidden_dim)

    def forward(self, x, h_prev):
        combined = torch.cat([x, h_prev], dim=1)
        r = torch.sigmoid(self.reset_gate(combined))   # reset: how much past to use
        z = torch.sigmoid(self.update_gate(combined))  # update: interpolation weight
        n = torch.tanh(self.new_gate(torch.cat([x, r * h_prev], dim=1)))
        h = (1 - z) * h_prev + z * n   # interpolate old and new
        return h

# Bidirectional LSTM: process sequence both forwards and backwards
bilstm = nn.LSTM(10, 128, batch_first=True, bidirectional=True)
x = torch.randn(32, 50, 10)
out, _ = bilstm(x)
print(f"BiLSTM output: {out.shape}")   # (32, 50, 256) — 2×128

Sequence Classification Example

import torch, torch.nn as nn

class SentimentClassifier(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, n_classes):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.lstm = nn.LSTM(embed_dim, hidden_dim, num_layers=2,
                            batch_first=True, dropout=0.3, bidirectional=True)
        self.dropout = nn.Dropout(0.5)
        self.fc = nn.Linear(2 * hidden_dim, n_classes)   # *2 for bidirectional

    def forward(self, token_ids, lengths):
        # token_ids: (batch, max_seq_len)
        embedded = self.dropout(self.embedding(token_ids))

        from torch.nn.utils.rnn import pack_padded_sequence, pad_packed_sequence
        packed = pack_padded_sequence(embedded, lengths.cpu(), batch_first=True,
                                      enforce_sorted=False)
        out, (h_n, _) = self.lstm(packed)

        # Concatenate last forward and backward hidden states
        h_fwd = h_n[-2]   # last layer, forward
        h_bwd = h_n[-1]   # last layer, backward
        h = torch.cat([h_fwd, h_bwd], dim=1)

        return self.fc(self.dropout(h))

model = SentimentClassifier(vocab_size=10000, embed_dim=128,
                             hidden_dim=256, n_classes=3)
token_ids = torch.randint(1, 10000, (32, 100))   # (batch, max_len)
lengths   = torch.randint(20, 100, (32,))
print(model(token_ids, lengths).shape)   # (32, 3)
