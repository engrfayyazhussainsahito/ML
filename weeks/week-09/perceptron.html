Lesson 9.1: The Perceptron and Activation Functions

The Biological Metaphor

A biological neuron receives signals through dendrites, integrates them in the cell body, and fires an output along the axon if the integrated signal exceeds a threshold. The artificial perceptron (Rosenblatt, 1957) captures this abstraction: it computes a weighted sum of inputs, adds a bias, and applies an activation function to produce an output.

y = f(w₁x₁ + w₂x₂ + … + wₙxₙ + b) = f(w·x + b)

The Perceptron Algorithm

import numpy as np

class Perceptron:
    """Single-layer binary perceptron with step activation."""
    def __init__(self, n_features, lr=0.1, n_epochs=50):
        self.w = np.zeros(n_features)
        self.b = 0.0
        self.lr = lr
        self.n_epochs = n_epochs

    def predict(self, X):
        return (X @ self.w + self.b >= 0).astype(int)

    def fit(self, X, y):
        for epoch in range(self.n_epochs):
            errors = 0
            for xi, yi in zip(X, y):
                y_hat = int(xi @ self.w + self.b >= 0)
                if y_hat != yi:
                    # Update rule: w += lr * (y - y_hat) * x
                    self.w += self.lr * (yi - y_hat) * xi
                    self.b += self.lr * (yi - y_hat)
                    errors += 1
            if errors == 0:
                print(f"Converged at epoch {epoch+1}"); break
        return self

# Perceptron theorem: converges in finite steps if data is linearly separable
from sklearn.datasets import make_classification
X, y = make_classification(n_samples=200, n_features=2, n_redundant=0,
                            n_clusters_per_class=1, random_state=42)
p = Perceptron(n_features=2, lr=0.1, n_epochs=100)
p.fit(X, y)
print(f"Accuracy: {(p.predict(X) == y).mean():.3f}")

The XOR Problem and Multi-Layer Networks

The perceptron can only learn linearly separable functions. XOR is not linearly separable — no single line can separate (0,0),(1,1) from (0,1),(1,0). Minsky and Papert (1969) proved this limitation, causing the first "AI winter". The solution: stack multiple layers, with non-linear activations between them.

import numpy as np

# XOR data
X = np.array([[0,0],[0,1],[1,0],[1,1]], dtype=float)
y = np.array([0, 1, 1, 0])

# Single perceptron: cannot solve XOR
p = Perceptron(n_features=2)
p.fit(X, y)
print("Perceptron on XOR:", p.predict(X))   # Will be wrong on ≥1 example

# Two-layer network: CAN solve XOR
import torch, torch.nn as nn

model = nn.Sequential(
    nn.Linear(2, 4),   # hidden layer: 4 neurons
    nn.ReLU(),
    nn.Linear(4, 1),   # output layer
    nn.Sigmoid()
)
opt = torch.optim.Adam(model.parameters(), lr=0.05)
X_t = torch.FloatTensor(X)
y_t = torch.FloatTensor(y).unsqueeze(1)

for _ in range(2000):
    opt.zero_grad()
    loss = nn.BCELoss()(model(X_t), y_t)
    loss.backward(); opt.step()

with torch.no_grad():
    preds = (model(X_t) > 0.5).float()
print("Two-layer network XOR:", preds.squeeze().numpy())  # [0,1,1,0] ✓

Activation Functions in Depth

import numpy as np, torch

def compare_activations(x):
    x_t = torch.tensor(x, dtype=torch.float32)
    results = {
        'sigmoid':      torch.sigmoid(x_t),
        'tanh':         torch.tanh(x_t),
        'relu':         torch.relu(x_t),
        'leaky_relu':   torch.nn.functional.leaky_relu(x_t, 0.01),
        'elu':          torch.nn.functional.elu(x_t),
        'gelu':         torch.nn.functional.gelu(x_t),
        'swish':        x_t * torch.sigmoid(x_t),
    }
    print(f"{'x':>8}", end='')
    for name in results: print(f"  {name:>12}", end='')
    print()
    for i, xi in enumerate(x):
        print(f"{xi:8.1f}", end='')
        for vals in results.values(): print(f"  {vals[i].item():12.4f}", end='')
        print()
