Lesson 15.1: Reinforcement Learning Primer

The RL Framework

Reinforcement learning formalises sequential decision making via the Markov Decision Process (MDP): an agent observes state s, takes action a, receives reward r, and transitions to next state s'. The goal is to learn a policy π(a|s) maximising expected discounted return G = Σ γᵗrₜ, where γ∈[0,1] is a discount factor penalising future rewards.

Q-Learning

import numpy as np, gymnasium as gym

env = gym.make("FrozenLake-v1", is_slippery=False)
n_states  = env.observation_space.n   # 16 grid cells
n_actions = env.action_space.n        # 4: left, down, right, up

# Q-table: Q(s, a) = expected return from state s taking action a then following policy
Q = np.zeros((n_states, n_actions))

alpha   = 0.8    # learning rate
gamma   = 0.95   # discount factor
epsilon = 1.0    # exploration rate (starts high, decays)

for episode in range(2000):
    state, _ = env.reset()
    done = False

    while not done:
        # Epsilon-greedy: explore with probability epsilon
        if np.random.rand() < epsilon:
            action = env.action_space.sample()
        else:
            action = Q[state].argmax()

        next_state, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated

        # Bellman update: Q(s,a) <- Q(s,a) + alpha*(r + gamma*max_a'Q(s',a') - Q(s,a))
        target = reward + gamma * Q[next_state].max() * (not terminated)
        Q[state, action] += alpha * (target - Q[state, action])
        state = next_state

    epsilon = max(0.01, epsilon * 0.995)   # decay exploration

# Evaluate
wins = sum(env.reset()[0] == 0 or True for _ in range(100))
print(f"Q-table learned. Optimal policy:")
for row in Q.argmax(axis=1).reshape(4,4):
    print(["←","↓","→","↑"][a] for a in row)

Deep Q-Network (DQN)

import torch, torch.nn as nn
from collections import deque
import random

class DQN(nn.Module):
    def __init__(self, state_dim, n_actions, hidden=128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden), nn.ReLU(),
            nn.Linear(hidden, hidden),    nn.ReLU(),
            nn.Linear(hidden, n_actions)
        )
    def forward(self, x): return self.net(x)

class ReplayBuffer:
    """Experience replay: decorrelates training samples."""
    def __init__(self, capacity=10000):
        self.buffer = deque(maxlen=capacity)

    def push(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)
        return (torch.FloatTensor(states), torch.LongTensor(actions),
                torch.FloatTensor(rewards), torch.FloatTensor(next_states),
                torch.FloatTensor(dones))

def dqn_update(online_net, target_net, buffer, opt, batch=64, gamma=0.99):
    if len(buffer.buffer) < batch: return
    s, a, r, s_, done = buffer.sample(batch)
    q_vals  = online_net(s).gather(1, a.unsqueeze(1)).squeeze()
    with torch.no_grad():
        q_next = target_net(s_).max(1).values
    target  = r + gamma * q_next * (1 - done)
    loss    = nn.SmoothL1Loss()(q_vals, target)
    opt.zero_grad(); loss.backward(); opt.step()

Policy Gradient Methods

import torch, torch.nn as nn
import gymnasium as gym

class PolicyNetwork(nn.Module):
    def __init__(self, state_dim, n_actions):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 128), nn.ReLU(),
            nn.Linear(128, n_actions)
        )
    def forward(self, x):
        return torch.softmax(self.net(x), dim=-1)

def reinforce(policy, env, optimiser, n_episodes=1000, gamma=0.99):
    for episode in range(n_episodes):
        states, actions, rewards = [], [], []
        state, _ = env.reset()
        done = False

        while not done:
            probs  = policy(torch.FloatTensor(state))
            action = torch.distributions.Categorical(probs).sample()
            next_state, reward, terminated, truncated, _ = env.step(action.item())
            states.append(state); actions.append(action); rewards.append(reward)
            state = next_state; done = terminated or truncated

        # Compute discounted returns
        returns = []
        G = 0
        for r in reversed(rewards):
            G = r + gamma * G; returns.insert(0, G)
        returns = torch.tensor(returns)
        returns = (returns - returns.mean()) / (returns.std() + 1e-8)   # normalise

        # Policy gradient update
        probs_all = policy(torch.FloatTensor(states))
        log_probs = torch.log(probs_all.gather(1, torch.tensor(actions).unsqueeze(1)).squeeze())
        loss = -(log_probs * returns).mean()
        optimiser.zero_grad(); loss.backward(); optimiser.step()
