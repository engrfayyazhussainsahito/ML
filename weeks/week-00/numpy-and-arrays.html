Lesson 0.2: NumPy & Arrays

Why NumPy?

Pure Python loops over lists are too slow for ML. A dataset of 1 million samples with 100 features requires 100 million operations per forward pass. NumPy solves this by storing arrays in contiguous memory blocks and dispatching to optimised C/FORTRAN routines, achieving 100–1000× speedups over Python loops.

Every major ML library — scikit-learn, PyTorch, TensorFlow, JAX — operates on NumPy-compatible array protocols. Mastering NumPy is mastering the substrate of all numerical ML.

Creating Arrays

import numpy as np

# From lists
a = np.array([1, 2, 3, 4])          # 1-D, shape (4,)
A = np.array([[1,2],[3,4],[5,6]])    # 2-D, shape (3,2)

print(A.shape)   # (3, 2)
print(A.ndim)    # 2
print(A.dtype)   # int64
print(A.size)    # 6

# Common constructors
zeros    = np.zeros((3, 4))
ones     = np.ones((2, 2))
eye      = np.eye(4)
rand     = np.random.randn(100, 10)
arange   = np.arange(0, 10, 0.5)
linspace = np.linspace(0, 1, 50)

Indexing and Slicing

X = np.random.randn(100, 5)

# Basic slicing (VIEW — no copy)
first_row   = X[0]           # shape (5,)
first_col   = X[:, 0]        # shape (100,)
sub_matrix  = X[10:20, 1:3]  # rows 10-19, cols 1-2

# Boolean indexing (COPY)
mask     = X[:, 0] > 0
positive = X[mask]            # rows where feature 0 is positive

# Fancy indexing
rows     = [0, 5, 10, 99]
selected = X[rows]            # shape (4, 5)

# In-place clipping
X[X < -3] = -3

Vectorised Operations and Broadcasting

Operations on NumPy arrays are element-wise and execute in C. Broadcasting automatically expands array dimensions to make shapes compatible — no loops needed.

a = np.array([1., 2., 3.])
b = np.array([10., 20., 30.])
print(a + b)          # [11. 22. 33.]
print(a * b)          # [10. 40. 90.]
print(np.exp(a))      # [2.718 7.389 20.09]

# Broadcasting: (3,1) op (1,4) -> (3,4)
col = np.array([[1],[2],[3]])       # shape (3, 1)
row = np.array([[10, 20, 30, 40]])  # shape (1, 4)
print((col + row).shape)            # (3, 4)

# Feature normalisation — no loop
X = np.random.randn(1000, 20)
mu  = X.mean(axis=0)   # shape (20,)
sig = X.std(axis=0)
X_norm = (X - mu) / sig
print(X_norm.mean(axis=0).round(8))   # all ~ 0
print(X_norm.std(axis=0).round(8))    # all ~ 1

Essential Matrix Operations

A = np.random.randn(50, 20)
w = np.random.randn(20)

# Matrix-vector product: all 50 predictions at once
y_hat = A @ w              # shape (50,)

# Matrix-matrix product
B = np.random.randn(20, 10)
C = A @ B                  # shape (50, 10)

# Transpose
print(A.T.shape)           # (20, 50)

# Linear system: Ax = b  (faster than inv(A) @ b)
A_sq = np.random.randn(10, 10)
b    = np.random.randn(10)
x    = np.linalg.solve(A_sq, b)
print(np.allclose(A_sq @ x, b))   # True

# SVD: used in PCA and recommender systems
U, S, Vt = np.linalg.svd(A, full_matrices=False)
print(U.shape, S.shape, Vt.shape)  # (50,20) (20,) (20,20)

# Reconstruct A from SVD (rank-k approximation)
k = 5
A_approx = U[:, :k] @ np.diag(S[:k]) @ Vt[:k, :]
print(np.linalg.norm(A - A_approx))  # reconstruction error

Aggregations Along Axes

X = np.random.randn(100, 5)

# Axis 0 = collapse rows (per-column result)
print(X.mean(axis=0).shape)    # (5,) — mean of each feature
print(X.sum(axis=0).shape)     # (5,)

# Axis 1 = collapse columns (per-row result)
print(X.sum(axis=1).shape)     # (100,) — row sums

# argmax/argmin
print(X.argmax(axis=1))        # index of max feature per sample

# keepdims prevents broadcasting bugs
mu = X.mean(axis=0, keepdims=True)  # (1,5) not (5,)
centered = X - mu

Performance: Always Vectorise

import time

n = 1_000_000
a = np.random.randn(n)
b = np.random.randn(n)

start = time.time()
result_py = sum(x*y for x, y in zip(a, b))
t_py = time.time() - start

start = time.time()
result_np = np.dot(a, b)
t_np = time.time() - start

print(f"Python loop: {t_py:.3f}s")
print(f"NumPy:       {t_np:.4f}s")
print(f"Speedup:     {t_py/t_np:.0f}x")
