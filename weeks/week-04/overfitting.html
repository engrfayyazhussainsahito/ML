Lesson 4.3: Overfitting and Underfitting

Diagnosing Model Fit

Overfitting and underfitting are the two fundamental failure modes in ML. An underfit model has not captured the true signal — it performs poorly on both training and test data (high bias). An overfit model has memorised noise — it performs well on training data but poorly on unseen data (high variance).

import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression, Ridge
from sklearn.pipeline import make_pipeline
from sklearn.metrics import mean_squared_error
from sklearn.model_selection import train_test_split

np.random.seed(42)
n = 60
x = np.linspace(0, 1, n).reshape(-1, 1)
y_true = np.sin(2 * np.pi * x.ravel())
y = y_true + np.random.randn(n) * 0.3

X_train, X_test, y_train, y_test = train_test_split(x, y, test_size=0.33)

for degree in [1, 3, 9, 15]:
    model = make_pipeline(PolynomialFeatures(degree), LinearRegression())
    model.fit(X_train, y_train)
    tr = np.sqrt(mean_squared_error(y_train, model.predict(X_train)))
    te = np.sqrt(mean_squared_error(y_test,  model.predict(X_test)))
    gap = te - tr
    status = "underfit" if degree==1 else "overfit" if degree>=9 else "good"
    print(f"deg={degree:2d}: train={tr:.3f}, test={te:.3f}, gap={gap:.3f} [{status}]")

Learning Curves

A learning curve plots training and validation error as a function of training set size. It is the most powerful diagnostic tool for identifying high bias vs high variance.

from sklearn.model_selection import learning_curve
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.datasets import make_classification

X, y = make_classification(n_samples=2000, n_features=20, random_state=42)

def plot_learning_curve(model, X, y, title="Learning Curve"):
    train_sizes, train_scores, val_scores = learning_curve(
        model, X, y,
        train_sizes=np.linspace(0.05, 1.0, 15),
        cv=5, scoring='accuracy', n_jobs=-1)

    train_mean = train_scores.mean(axis=1)
    train_std  = train_scores.std(axis=1)
    val_mean   = val_scores.mean(axis=1)
    val_std    = val_scores.std(axis=1)

    print(f"\n{title}")
    print(f"{'Size':>6}  {'Train':>6}  {'Val':>6}  {'Gap':>6}")
    for sz, tr, va in zip(train_sizes, train_mean, val_mean):
        print(f"{sz:6.0f}  {tr:.4f}  {va:.4f}  {tr-va:.4f}")

# High-variance model (deep forest, no regularisation)
plot_learning_curve(
    RandomForestClassifier(n_estimators=200, max_depth=None),
    X, y, "Deep Random Forest (high variance)")

# Low-variance model
plot_learning_curve(
    RandomForestClassifier(n_estimators=50, max_depth=3, min_samples_leaf=20),
    X, y, "Shallow Forest (more regularised)")
PatternDiagnosisFix
Both curves converge at HIGH errorHigh bias (underfitting)More complex model, more features, less regularisation
Large GAP between train and valHigh variance (overfitting)More data, regularisation, simpler model, dropout
Curves converge at LOW errorGood fitConsider collecting more data anyway

Regularisation Techniques

import torch, torch.nn as nn

class RegularisedMLP(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim, dropout_p=0.3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.BatchNorm1d(hidden_dim),   # batch norm stabilises training
            nn.ReLU(),
            nn.Dropout(dropout_p),        # randomly zero neurons during training
            nn.Linear(hidden_dim, hidden_dim),
            nn.BatchNorm1d(hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout_p),
            nn.Linear(hidden_dim, output_dim)
        )

    def forward(self, x):
        return self.net(x)

# L2 regularisation via weight_decay in the optimiser
model = RegularisedMLP(20, 128, 1)
optimiser = torch.optim.Adam(model.parameters(), lr=1e-3, weight_decay=1e-4)
