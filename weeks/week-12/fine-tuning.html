Lesson 12.4: Fine-Tuning and Prompt Engineering

Adapting Foundation Models

Foundation models contain billions of parameters trained on trillions of tokens. Adapting them to downstream tasks involves tradeoffs between performance, cost, and data requirements. The key techniques are full fine-tuning, parameter-efficient fine-tuning (PEFT), and prompt engineering.

Full Fine-Tuning

from transformers import (AutoModelForSequenceClassification, AutoTokenizer,
                          TrainingArguments, Trainer)
from datasets import load_dataset
import numpy as np

model_name = "distilbert-base-uncased"
tokeniser  = AutoTokenizer.from_pretrained(model_name)
model      = AutoModelForSequenceClassification.from_pretrained(model_name, num_labels=2)

dataset = load_dataset("imdb")

def tokenise(batch):
    return tokeniser(batch["text"], truncation=True, max_length=256)

tokenised = dataset.map(tokenise, batched=True)

training_args = TrainingArguments(
    output_dir="./results",
    num_train_epochs=3,
    per_device_train_batch_size=16,
    per_device_eval_batch_size=32,
    learning_rate=2e-5,
    weight_decay=0.01,
    warmup_ratio=0.1,
    evaluation_strategy="epoch",
    load_best_model_at_end=True,
    metric_for_best_model="accuracy",
)

def compute_metrics(eval_pred):
    logits, labels = eval_pred
    preds = np.argmax(logits, axis=1)
    return {"accuracy": (preds == labels).mean()}

trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenised["train"].select(range(2000)),
    eval_dataset=tokenised["test"].select(range(500)),
    compute_metrics=compute_metrics,
)
trainer.train()

LoRA: Low-Rank Adaptation

LoRA (Hu et al., 2021) freezes all pretrained weights and injects small trainable rank-decomposition matrices into each attention layer. Instead of updating a W∈ℝ^{d×k} matrix directly, it learns W + BA where B∈ℝ^{d×r}, A∈ℝ^{r×k}, with r<<d. This reduces trainable parameters by 10,000× while matching full fine-tuning performance.

from peft import LoraConfig, get_peft_model, TaskType

lora_config = LoraConfig(
    task_type=TaskType.SEQ_CLS,
    r=8,             # rank of decomposition matrices
    lora_alpha=16,   # scaling factor: weight = (lora_alpha/r) * BA
    lora_dropout=0.1,
    target_modules=["q_lin", "k_lin", "v_lin"],  # which layers to adapt
)

model_lora = get_peft_model(model, lora_config)
model_lora.print_trainable_parameters()
# trainable params: ~300K || all params: 66M || trainable%: 0.45%
# Full fine-tuning would update all 66M params — LoRA: only 300K

Prompt Engineering

from transformers import pipeline

# Zero-shot classification
classifier = pipeline("zero-shot-classification",
                       model="facebook/bart-large-mnli")

text = "This quarterly report shows declining revenue in the APAC region."
candidate_labels = ["finance", "technology", "sports", "politics"]

result = classifier(text, candidate_labels)
print(result['labels'])   # ['finance', 'technology', ...]
print(result['scores'])   # [0.91, 0.04, ...]

# Few-shot prompting for GPT-style models
few_shot_prompt = """
Classify the sentiment as positive or negative.

Text: The food was amazing!
Sentiment: positive

Text: Service was incredibly slow.
Sentiment: negative

Text: {new_text}
Sentiment:"""

new_text = "The delivery arrived on time but packaging was damaged."
prompt = few_shot_prompt.format(new_text=new_text)
