Lesson 6.2: Hierarchical Clustering

Hierarchical Clustering

Hierarchical clustering builds a tree of clusters (dendrogram) without requiring k to be specified in advance. Agglomerative (bottom-up) clustering starts with each point as its own cluster and merges the closest pair at each step. Divisive (top-down) starts with one cluster and recursively splits.

from sklearn.cluster import AgglomerativeClustering
from sklearn.datasets import make_blobs
from sklearn.preprocessing import StandardScaler
from scipy.cluster.hierarchy import dendrogram, linkage
import numpy as np

X, y_true = make_blobs(n_samples=150, centers=4, random_state=42)
X = StandardScaler().fit_transform(X)

# Agglomerative clustering with ward linkage
agg = AgglomerativeClustering(n_clusters=4, linkage='ward')
labels = agg.fit_predict(X)

from sklearn.metrics import adjusted_rand_score
print(f"ARI: {adjusted_rand_score(y_true, labels):.4f}")

Linkage Criteria

LinkageDistance between clusters A, BBehaviour
Singlemin distance between any pairChaining — forms elongated clusters
Completemax distance between any pairCompact, similar-diameter clusters
Averagemean distance between all pairsCompromise between single/complete
Wardincrease in total within-cluster varianceSpherical, equal-size clusters — usually best
 
from scipy.cluster.hierarchy import linkage, dendrogram, fcluster
import numpy as np

# Linkage matrix: encodes the full merging history
Z = linkage(X, method='ward')
# Z[i] = [cluster_a, cluster_b, distance, n_samples_in_new_cluster]
print(f"Linkage matrix shape: {Z.shape}")  # (n-1, 4)

# Cut dendrogram at different heights to get different k
for k in [2, 3, 4, 5]:
    labels = fcluster(Z, k, criterion='maxclust')
    from sklearn.metrics import silhouette_score
    sil = silhouette_score(X, labels)
    print(f"k={k}: silhouette={sil:.4f}")

Reading a Dendrogram

from scipy.cluster.hierarchy import dendrogram, linkage
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

Z = linkage(X[:30], method='ward')   # small subset for readability

plt.figure(figsize=(10, 4))
dendrogram(Z, truncate_mode='lastp', p=10,
           show_leaf_counts=True, leaf_rotation=45)
plt.xlabel("Cluster (sample count)")
plt.ylabel("Distance (Ward)")
plt.title("Dendrogram — Ward Linkage")
plt.tight_layout()
plt.savefig("/tmp/dendrogram.png", dpi=80)

# The height of a merge = distance between the two clusters
# Large jumps suggest natural cluster boundaries
# Cut horizontally at a large gap to choose k
