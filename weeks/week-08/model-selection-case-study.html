Lesson 8.3: Model Selection Case Study

The Model Selection Problem

Choosing the best model from several candidates is a statistical decision, not just a ranking by CV score. A difference of 0.3% accuracy between two models on the same dataset may be noise. We need statistical tests to distinguish real differences from sampling variation, and we need nested evaluation to avoid selection bias.

Systematic Comparison

from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.svm import SVC
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import cross_validate, StratifiedKFold
import numpy as np, pandas as pd
from sklearn.datasets import load_breast_cancer

X, y = load_breast_cancer(return_X_y=True)

models = {
    'LogReg': Pipeline([('s', StandardScaler()), ('m', LogisticRegression(max_iter=1000))]),
    'RF':     RandomForestClassifier(n_estimators=200, random_state=42),
    'GBM':    GradientBoostingClassifier(n_estimators=300, random_state=42),
    'SVM':    Pipeline([('s', StandardScaler()), ('m', SVC(probability=True))]),
}

cv = StratifiedKFold(n_splits=10, shuffle=True, random_state=42)
results = {}

for name, model in models.items():
    res = cross_validate(model, X, y, cv=cv,
                         scoring=['accuracy','roc_auc','f1'],
                         return_train_score=True, n_jobs=-1)
    results[name] = {
        'accuracy_mean': res['test_accuracy'].mean(),
        'accuracy_std':  res['test_accuracy'].std(),
        'auc_mean':      res['test_roc_auc'].mean(),
        'f1_mean':       res['test_f1'].mean(),
        'train_acc':     res['train_accuracy'].mean(),
        'cv_scores':     res['test_accuracy'],  # keep for stats tests
    }

df_results = pd.DataFrame(results).T
print(df_results[['accuracy_mean','accuracy_std','auc_mean','f1_mean']].round(4))

Statistical Significance Testing

from scipy.stats import wilcoxon
import numpy as np

# Wilcoxon signed-rank test: non-parametric, accounts for paired structure of CV
best_model = max(results, key=lambda n: results[n]['accuracy_mean'])
best_scores = results[best_model]['cv_scores']

print(f"Best model: {best_model}")
print(f"\nComparisons vs {best_model}:")
for name, res in results.items():
    if name == best_model: continue
    scores = res['cv_scores']
    stat, p = wilcoxon(best_scores, scores, alternative='greater')
    diff = (best_scores - scores).mean()
    sig = "**" if p < 0.01 else "*" if p < 0.05 else "ns"
    print(f"  vs {name:8s}: diff={diff:+.4f}, p={p:.4f} {sig}")

Bias–Variance Analysis of Models

import numpy as np

print("\nBias-Variance proxy (train_acc - val_acc = variance gap):")
for name, res in results.items():
    gap = res['train_acc'] - res['accuracy_mean']
    diag = "overfitting" if gap > 0.05 else "good"
    print(f"  {name:10s}: train={res['train_acc']:.4f}, "
          f"val={res['accuracy_mean']:.4f}, gap={gap:.4f} [{diag}]")
