Lesson 7.4: scikit-learn Pipelines

Why Pipelines?

Without a Pipeline, applying preprocessing steps manually before cross-validation causes data leakage — the scaler, imputer, or encoder sees validation data during fit, inflating performance estimates. Pipeline chains all steps, automatically fitting on training folds only and transforming validation folds. It also makes deployment trivial: one object processes raw data end-to-end.

Basic Pipeline

from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import cross_val_score
import numpy as np

X = np.random.randn(500, 10)
X[np.random.rand(*X.shape) < 0.1] = np.nan   # add missing values
y = np.random.randint(0, 2, 500)

# Steps: (name, estimator) pairs
pipe = Pipeline([
    ('impute', SimpleImputer(strategy='median')),   # 1. fill missing
    ('scale',  StandardScaler()),                   # 2. normalise
    ('clf',    RandomForestClassifier(n_estimators=100, random_state=42))  # 3. model
])

# Cross-validation: imputer and scaler fit only on train folds
scores = cross_val_score(pipe, X, y, cv=5, scoring='accuracy')
print(f"CV accuracy: {scores.mean():.4f} +/- {scores.std():.4f}")

# Fit on full training data, then predict
pipe.fit(X, y)
X_new = np.random.randn(5, 10)
print(pipe.predict(X_new))

# Access intermediate steps
print(pipe.named_steps['scale'].mean_[:5])

ColumnTransformer: Different Processing per Column

import pandas as pd
import numpy as np
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.model_selection import train_test_split

df = pd.DataFrame({
    'age':      [25, np.nan, 47, 29, 55, 33],
    'income':   [50000, 72000, np.nan, 45000, 90000, 61000],
    'city':     ['NYC', 'LA', 'NYC', 'SF', 'LA', 'NYC'],
    'employed': [1, 1, 0, 1, 1, 0],
    'default':  [0, 0, 1, 0, 0, 1]
})

num_cols = ['age', 'income']
cat_cols = ['city']
bin_cols = ['employed']
target   = 'default'

X = df.drop(columns=[target])
y = df[target]

# Numerical pipeline: impute then scale
num_pipe = Pipeline([
    ('impute', SimpleImputer(strategy='median')),
    ('scale',  StandardScaler())
])

# Categorical pipeline: impute mode then one-hot
cat_pipe = Pipeline([
    ('impute', SimpleImputer(strategy='most_frequent')),
    ('ohe',    OneHotEncoder(handle_unknown='ignore', sparse_output=False))
])

preprocessor = ColumnTransformer([
    ('num', num_pipe, num_cols),
    ('cat', cat_pipe, cat_cols),
    ('bin', 'passthrough', bin_cols),
])

full_pipe = Pipeline([
    ('preprocess', preprocessor),
    ('clf', GradientBoostingClassifier(random_state=42))
])

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.33)
full_pipe.fit(X_train, y_train)
print(f"Test accuracy: {full_pipe.score(X_test, y_test):.4f}")

Hyperparameter Tuning Through a Pipeline

from sklearn.model_selection import RandomizedSearchCV
from scipy.stats import randint, loguniform

# Reference pipeline steps via double underscore
param_distributions = {
    'clf__n_estimators':     randint(50, 300),
    'clf__learning_rate':    loguniform(1e-3, 0.3),
    'clf__max_depth':        randint(2, 8),
    'preprocess__num__scale': [StandardScaler(), 'passthrough'],
}

search = RandomizedSearchCV(
    full_pipe, param_distributions,
    n_iter=30, cv=5, scoring='accuracy',
    random_state=42, n_jobs=-1
)
search.fit(X_train, y_train)
print(f"Best params: {search.best_params_}")
print(f"Best CV score: {search.best_score_:.4f}")
