Lesson 2.3: Regularisation

The Overfitting Problem

A high-degree polynomial can fit training data perfectly while performing terribly on new data — it memorises noise rather than the true signal. Regularisation adds a penalty term to the loss function that discourages large weights, controlling model complexity implicitly.

import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression, Ridge, Lasso
from sklearn.pipeline import make_pipeline
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

np.random.seed(42)
x = np.linspace(0, 1, 30).reshape(-1, 1)
y = np.sin(2 * np.pi * x.ravel()) + np.random.randn(30) * 0.3

X_train, X_test, y_train, y_test = train_test_split(x, y, test_size=0.33)

for degree in [1, 3, 9, 15]:
    model = make_pipeline(PolynomialFeatures(degree), LinearRegression())
    model.fit(X_train, y_train)
    train_rmse = np.sqrt(mean_squared_error(y_train, model.predict(X_train)))
    test_rmse  = np.sqrt(mean_squared_error(y_test,  model.predict(X_test)))
    print(f"Degree {degree:2d}: train={train_rmse:.3f}, test={test_rmse:.3f}")
# Degree 15: train≈0.01, test>>1 — extreme overfitting

L2 Regularisation: Ridge Regression

Ridge adds the squared L2 norm of weights to the loss:

L_ridge(w) = ||y - Xw||² + α||w||²

The hyperparameter α controls the strength of regularisation. α→0 recovers OLS; α→∞ shrinks all weights to zero. Ridge has a closed-form solution: w* = (XᵀX + αI)⁻¹Xᵀy, which is always invertible even with collinear features.

from sklearn.linear_model import Ridge, RidgeCV
import numpy as np

# RidgeCV finds the best alpha via leave-one-out cross-validation
alphas = np.logspace(-4, 4, 100)
ridge_cv = RidgeCV(alphas=alphas, cv=5)
ridge_cv.fit(X_train, y_train)
print(f"Best alpha: {ridge_cv.alpha_:.4f}")
print(f"Test RMSE:  {np.sqrt(mean_squared_error(y_test, ridge_cv.predict(X_test))):.3f}")

# Effect of alpha on coefficients
from sklearn.linear_model import ridge_regression
for alpha in [0.001, 1.0, 100.0]:
    w = ridge_regression(X_train, y_train, alpha=alpha)
    print(f"alpha={alpha:7.3f}  ||w||={np.linalg.norm(w):.3f}")
# As alpha increases, ||w|| decreases

L1 Regularisation: Lasso

Lasso adds the L1 norm of weights:

L_lasso(w) = ||y - Xw||² + α||w||₁

Lasso performs automatic feature selection: its L1 penalty drives many weights exactly to zero (sparse solutions), unlike Ridge which only shrinks weights toward (but not to) zero. Use Lasso when you suspect only a few features are truly predictive.

from sklearn.linear_model import Lasso, LassoCV
import numpy as np

# High-dimensional problem: 200 samples, 500 features, only 10 are relevant
n, p = 200, 500
X = np.random.randn(n, p)
true_w = np.zeros(p)
true_w[:10] = np.random.randn(10) * 3   # only first 10 features matter
y = X @ true_w + np.random.randn(n) * 0.5

lasso_cv = LassoCV(cv=5, random_state=42)
lasso_cv.fit(X, y)

# Count non-zero coefficients
n_nonzero = np.sum(lasso_cv.coef_ != 0)
print(f"Alpha: {lasso_cv.alpha_:.4f}")
print(f"Non-zero coefficients: {n_nonzero} (true: 10)")
print(f"Recovered true features: {np.sum(np.abs(lasso_cv.coef_[:10]) > 0.1)}")

Elastic Net: Best of Both

Elastic Net combines L1 and L2 penalties:

L_enet(w) = ||y - Xw||² + α(ρ||w||₁ + (1-ρ)||w||²)

The mixing parameter ρ (l1_ratio in sklearn) interpolates between Lasso (ρ=1) and Ridge (ρ=0). Elastic Net is preferred when features are correlated and you want both sparsity and stability.

from sklearn.linear_model import ElasticNetCV
import numpy as np

enet = ElasticNetCV(
    l1_ratio=[0.1, 0.5, 0.7, 0.9, 0.95, 1.0],
    alphas=np.logspace(-4, 1, 50),
    cv=5
)
enet.fit(X_train, y_train)
print(f"Best alpha:    {enet.alpha_:.4f}")
print(f"Best l1_ratio: {enet.l1_ratio_:.2f}")
print(f"Non-zero:      {np.sum(enet.coef_ != 0)}")

Regularisation Path

from sklearn.linear_model import lasso_path
import numpy as np, matplotlib.pyplot as plt

alphas, coefs, _ = lasso_path(X_train, y_train, n_alphas=100)

plt.figure(figsize=(8, 4))
plt.semilogx(alphas, coefs.T)
plt.xlabel("Alpha (log scale)")
plt.ylabel("Coefficient value")
plt.title("Lasso regularisation path")
plt.axvline(lasso_cv.alpha_, color='k', linestyle='--', label="CV best")
plt.legend(); plt.gca().invert_xaxis()   # high alpha = more regularisation
plt.show()
