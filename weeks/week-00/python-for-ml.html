Lesson 0.1: Python for ML

Why Python for ML?

Python has become the lingua franca of machine learning due to its readable syntax, interactive notebooks, and a rich ecosystem of scientific libraries. Understanding Python well is a prerequisite for every subsequent lesson.

You do not need to master all of Python before doing ML. This lesson covers the subset you will actually use: data types, control flow, functions, classes, and the libraries NumPy, pandas, and matplotlib.

Core Data Structures

# Lists — ordered, mutable sequences
features = [1.5, 2.3, 0.8, 4.1]
features.append(3.0)
print(features[0])     # 1.5
print(features[-1])    # 3.0 (last element)

# Dictionaries — key-value stores
sample = {"age": 34, "income": 72000, "label": 1}
print(sample["age"])   # 34

# List comprehensions — concise loops
squares = [x**2 for x in range(10)]
evens   = [x for x in range(20) if x % 2 == 0]

# Tuple unpacking
x, y, z = 1.0, 2.0, 3.0

Functions and Lambda

def sigmoid(x):
    """Compute the sigmoid activation function."""
    import math
    return 1 / (1 + math.exp(-x))

print(sigmoid(0))    # 0.5
print(sigmoid(2))    # 0.88...

# Lambda — anonymous one-liner functions
square = lambda x: x ** 2

# Functions as arguments (higher-order functions)
data = [3, 1, 4, 1, 5, 9, 2, 6]
sorted_data = sorted(data, key=lambda x: -x)   # descending

Classes for ML Models

class KNNClassifier:
    """A minimal k-nearest neighbours classifier."""

    def __init__(self, k=3):
        self.k = k
        self.X_train = None
        self.y_train = None

    def fit(self, X, y):
        """Store training data."""
        self.X_train = X
        self.y_train = y
        return self

    def predict(self, X):
        """Predict class for each row in X."""
        import numpy as np
        predictions = []
        for x in X:
            # Euclidean distances to all training points
            dists = np.linalg.norm(self.X_train - x, axis=1)
            # k nearest neighbour indices
            nn_idx = np.argsort(dists)[:self.k]
            # Majority vote
            labels, counts = np.unique(self.y_train[nn_idx], return_counts=True)
            predictions.append(labels[np.argmax(counts)])
        return np.array(predictions)
