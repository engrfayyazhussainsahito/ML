Lesson 10.3: Transfer Learning

The Transfer Learning Principle

A CNN trained on ImageNet (1.2M images, 1000 classes) learns general visual features in its early layers — edges, textures, shapes — that transfer to almost any vision task. Transfer learning uses this pre-learned knowledge rather than training from scratch, dramatically reducing data and compute requirements.

Feature Extraction vs Fine-Tuning

Feature ExtractionFull Fine-TuningPartial Fine-Tuning
Pretrained weightsFrozenAll updatedLater layers updated
Data neededVery little (100s)Moderate (10k+)Little (1k+)
Training speedFastSlowModerate
PerformanceGood baselineBest (if enough data)Usually best tradeoff
 
import torch, torch.nn as nn
import torchvision.models as models
import torchvision.transforms as T
from torch.utils.data import DataLoader

num_classes = 5   # e.g. flower species classification

# Strategy 1: Feature extraction — freeze all, replace head
model_fe = models.resnet50(weights=models.ResNet50_Weights.IMAGENET1K_V2)
for param in model_fe.parameters():
    param.requires_grad = False          # freeze all layers
model_fe.fc = nn.Linear(2048, num_classes)   # replace head only
# Only head parameters are trained — fast, few parameters

# Strategy 2: Full fine-tuning — unfreeze all
model_ft = models.resnet50(weights=models.ResNet50_Weights.IMAGENET1K_V2)
model_ft.fc = nn.Linear(2048, num_classes)
# All parameters trained — use small learning rate for pretrained layers

# Strategy 3: Partial fine-tuning — freeze early, train late
model_pft = models.resnet50(weights=models.ResNet50_Weights.IMAGENET1K_V2)
# Freeze everything
for param in model_pft.parameters():
    param.requires_grad = False
# Unfreeze layer3, layer4, and head
for layer in [model_pft.layer3, model_pft.layer4]:
    for param in layer.parameters():
        param.requires_grad = True
model_pft.fc = nn.Linear(2048, num_classes)

# Count trainable parameters
def count_trainable(m):
    return sum(p.numel() for p in m.parameters() if p.requires_grad)
print(f"Feature extraction: {count_trainable(model_fe):,} trainable params")
print(f"Partial fine-tuning: {count_trainable(model_pft):,} trainable params")
print(f"Full fine-tuning:   {count_trainable(model_ft):,} trainable params")

Training with Differential Learning Rates

import torch.optim as optim

model = model_pft  # partial fine-tuning model

# Different learning rates: smaller for pretrained layers, larger for head
optimiser = optim.AdamW([
    {'params': model.layer3.parameters(), 'lr': 1e-5},
    {'params': model.layer4.parameters(), 'lr': 3e-5},
    {'params': model.fc.parameters(),     'lr': 1e-3},
], weight_decay=1e-4)

# Learning rate scheduler: warmup + cosine decay
scheduler = optim.lr_scheduler.OneCycleLR(
    optimiser, max_lr=[1e-5, 3e-5, 1e-3],
    steps_per_epoch=100, epochs=20
)

Data Augmentation for Small Datasets

import torchvision.transforms as T

train_transforms = T.Compose([
    T.RandomResizedCrop(224, scale=(0.8, 1.0)),
    T.RandomHorizontalFlip(p=0.5),
    T.RandomRotation(degrees=15),
    T.ColorJitter(brightness=0.3, contrast=0.3, saturation=0.2, hue=0.1),
    T.RandomGrayscale(p=0.1),
    T.ToTensor(),
    T.Normalize(mean=[0.485, 0.456, 0.406],   # ImageNet mean/std
                std=[0.229, 0.224, 0.225])
])

val_transforms = T.Compose([
    T.Resize(256), T.CenterCrop(224),
    T.ToTensor(),
    T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])
