Lesson 14.3: Privacy and Data Governance

Privacy Risks in ML

ML models can leak information about their training data. Membership inference attacks determine whether a specific sample was in the training set. Model inversion attacks reconstruct training examples from model outputs. Attribute inference attacks recover sensitive attributes of training individuals. These are not hypothetical — they have been demonstrated against production models including GPT-2 and medical models.

Differential Privacy

import torch, torch.nn as nn
from opacus import PrivacyEngine
from opacus.validators import ModuleValidator

# Differential Privacy: add calibrated Gaussian noise to gradients during training
# Guarantee: changing one training example changes the output distribution by at most epsilon

model = nn.Sequential(nn.Linear(10, 64), nn.ReLU(), nn.Linear(64, 2))
# Some layers (BatchNorm) are incompatible with DP
model = ModuleValidator.fix(model)   # replaces BatchNorm with GroupNorm

optimiser = torch.optim.Adam(model.parameters(), lr=1e-3)
train_loader = torch.utils.data.DataLoader(
    torch.utils.data.TensorDataset(torch.randn(1000, 10), torch.randint(0, 2, (1000,))),
    batch_size=64
)

privacy_engine = PrivacyEngine()
model, optimiser, train_loader = privacy_engine.make_private_with_epsilon(
    module=model,
    optimizer=optimiser,
    data_loader=train_loader,
    epochs=10,
    target_epsilon=1.0,    # privacy budget: lower = stronger privacy
    target_delta=1e-5,     # probability of privacy failure
    max_grad_norm=1.0,     # gradient clipping magnitude
)

# Training loop is identical — Opacus handles noise addition automatically
for batch_x, batch_y in train_loader:
    optimiser.zero_grad()
    loss = nn.CrossEntropyLoss()(model(batch_x.float()), batch_y)
    loss.backward(); optimiser.step()

epsilon, delta = privacy_engine.get_epsilon(delta=1e-5), 1e-5
print(f"Privacy guarantee: ({epsilon:.2f}, {delta})-DP")

Federated Learning

import torch, torch.nn as nn, copy

def federated_avg(global_model, client_models, client_sizes):
    """
    FedAvg: weighted average of client model parameters.
    Clients train locally on private data; only parameters (not data) are shared.
    """
    total = sum(client_sizes)
    global_state = global_model.state_dict()

    for key in global_state:
        # Weighted average by number of local samples
        global_state[key] = sum(
            client_models[i].state_dict()[key] * (client_sizes[i] / total)
            for i in range(len(client_models))
        )
    global_model.load_state_dict(global_state)
    return global_model

def client_train(model, local_data, local_labels, n_epochs=5, lr=0.01):
    """Simulate local training on a client device."""
    local_model = copy.deepcopy(model)
    opt = torch.optim.SGD(local_model.parameters(), lr=lr)
    for _ in range(n_epochs):
        opt.zero_grad()
        loss = nn.CrossEntropyLoss()(local_model(local_data), local_labels)
        loss.backward(); opt.step()
    return local_model

# Simulate 3 clients
global_model = nn.Sequential(nn.Linear(10, 32), nn.ReLU(), nn.Linear(32, 2))
client_data  = [torch.randn(100, 10) for _ in range(3)]
client_labels= [torch.randint(0, 2, (100,)) for _ in range(3)]
client_sizes = [100, 100, 100]

for round_num in range(5):
    client_models = [client_train(global_model, X, y)
                     for X, y in zip(client_data, client_labels)]
    global_model  = federated_avg(global_model, client_models, client_sizes)
    print(f"Round {round_num+1} complete")
