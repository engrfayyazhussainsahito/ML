Lesson 3.1: Logistic Regression

From Regression to Classification

Linear regression outputs any real number — unsuitable for probabilities, which must lie in [0,1]. Logistic regression applies the sigmoid function to a linear combination of features, producing a calibrated probability estimate. Despite the name, logistic regression is a classification algorithm.

import numpy as np

def sigmoid(z):
    """Numerically stable sigmoid: sigma(z) = 1 / (1 + e^(-z))"""
    return np.where(z >= 0,
                    1 / (1 + np.exp(-z)),
                    np.exp(z) / (1 + np.exp(z)))

z = np.array([-10, -2, 0, 2, 10])
print(sigmoid(z))
# [4.5e-05, 0.119, 0.500, 0.881, 1.000]
# z=0 -> P=0.5 (decision boundary)

The Log-Loss (Binary Cross-Entropy)

We train logistic regression by maximising the log-likelihood, equivalently minimising the binary cross-entropy loss:

L(w) = -(1/n) Σᵢ [yᵢ log(p̂ᵢ) + (1-yᵢ) log(1-p̂ᵢ)]

This loss has no closed form — we minimise it with gradient descent. The gradient is beautifully simple: ∂L/∂w = (1/n) Xᵀ(p̂ - y).

import numpy as np

def logistic_loss_and_grad(X, y, w, lam=0.0):
    """
    Binary cross-entropy + L2 regularisation.
    X: (n, p+1) with bias column; y: (n,); w: (p+1,)
    """
    n = len(y)
    p_hat = sigmoid(X @ w)
    # Clip to avoid log(0)
    p_hat = np.clip(p_hat, 1e-12, 1-1e-12)
    loss = -(y * np.log(p_hat) + (1-y) * np.log(1-p_hat)).mean()
    loss += 0.5 * lam * (w[1:]**2).sum()  # don't regularise bias

    grad = X.T @ (p_hat - y) / n
    grad[1:] += lam * w[1:]
    return loss, grad

def fit_logistic(X, y, lr=0.1, epochs=1000, lam=0.01):
    X_b = np.column_stack([np.ones(len(X)), X])
    w = np.zeros(X_b.shape[1])
    for epoch in range(epochs):
        loss, grad = logistic_loss_and_grad(X_b, y, w, lam)
        w -= lr * grad
        if epoch % 200 == 0:
            print(f"Epoch {epoch}: loss={loss:.4f}")
    return w

scikit-learn and the Decision Boundary

from sklearn.linear_model import LogisticRegression
from sklearn.datasets import make_classification
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.metrics import roc_auc_score
import numpy as np

X, y = make_classification(n_samples=1000, n_features=2,
                            n_informative=2, n_redundant=0,
                            random_state=42)

# C = 1/lambda: higher C = less regularisation
pipe = make_pipeline(
    StandardScaler(),
    LogisticRegression(C=1.0, solver='lbfgs', max_iter=1000)
)
pipe.fit(X, y)

# Probability outputs
proba = pipe.predict_proba(X)[:, 1]
print(f"ROC-AUC: {roc_auc_score(y, proba):.3f}")

# Predicted class at threshold 0.5
y_pred = (proba >= 0.5).astype(int)
print(f"Accuracy: {(y_pred == y).mean():.3f}")

Multi-class: Softmax

For K classes, we replace sigmoid with softmax. Softmax converts K raw scores (logits) into a probability distribution over classes:

P(y=k|x) = exp(wₖᵀx) / Σⱼ exp(wⱼᵀx)
import numpy as np

def softmax(logits):
    """Numerically stable softmax."""
    z = logits - logits.max(axis=-1, keepdims=True)  # subtract max for stability
    exp_z = np.exp(z)
    return exp_z / exp_z.sum(axis=-1, keepdims=True)

logits = np.array([[1.0, 2.0, 0.5, -1.0]])   # 4 classes
probs = softmax(logits)
print(probs)
print(probs.sum())   # 1.0

# sklearn multi-class logistic regression
from sklearn.linear_model import LogisticRegression
from sklearn.datasets import load_iris

X_iris, y_iris = load_iris(return_X_y=True)
lr_multi = LogisticRegression(multi_class='multinomial', max_iter=1000)
lr_multi.fit(X_iris, y_iris)
print(f"3-class accuracy: {lr_multi.score(X_iris, y_iris):.3f}")

Threshold Selection

from sklearn.metrics import precision_recall_curve, f1_score
import numpy as np

# Default threshold is 0.5 — rarely optimal
proba = pipe.predict_proba(X)[:, 1]

# Find threshold that maximises F1
precisions, recalls, thresholds = precision_recall_curve(y, proba)
f1s = 2 * precisions * recalls / (precisions + recalls + 1e-9)
best_threshold = thresholds[np.argmax(f1s[:-1])]
print(f"Best threshold: {best_threshold:.3f}")
print(f"Best F1:        {f1s.max():.3f}")

y_pred_opt = (proba >= best_threshold).astype(int)
