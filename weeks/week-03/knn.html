Lesson 3.3: K-Nearest Neighbours

The K-Nearest Neighbours Algorithm

KNN is a non-parametric, instance-based learner. Training simply stores the data. For a new point, it finds the k closest training examples and predicts by majority vote (classification) or average (regression). No explicit model parameters are learned — all computation happens at prediction time.

import numpy as np

def euclidean_distance(a, b):
    return np.sqrt(((a - b)**2).sum())

def knn_predict(X_train, y_train, x_new, k=5):
    """KNN prediction for a single point x_new."""
    dists = np.array([euclidean_distance(x_new, x) for x in X_train])
    nn_idx = np.argsort(dists)[:k]
    nn_labels = y_train[nn_idx]
    # Majority vote
    values, counts = np.unique(nn_labels, return_counts=True)
    return values[np.argmax(counts)]

# sklearn implementation (much faster — uses KD-tree or ball-tree)
from sklearn.neighbors import KNeighborsClassifier
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline

X, y = make_classification(n_samples=1000, n_features=10, random_state=42)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)

# ALWAYS scale before KNN — distances are not scale-invariant
knn = make_pipeline(StandardScaler(), KNeighborsClassifier(n_neighbors=5))
knn.fit(X_train, y_train)
print(f"Test accuracy: {knn.score(X_test, y_test):.3f}")

Choosing k

k is the critical hyperparameter. Small k (e.g. k=1) gives a complex, jagged decision boundary — low bias, high variance. Large k gives a smoother boundary — higher bias, lower variance. The optimal k balances this tradeoff and should be found via cross-validation.

from sklearn.model_selection import cross_val_score
import numpy as np

k_values = range(1, 50, 2)
cv_scores = []
for k in k_values:
    knn = make_pipeline(StandardScaler(), KNeighborsClassifier(n_neighbors=k))
    scores = cross_val_score(knn, X_train, y_train, cv=5, scoring='accuracy')
    cv_scores.append(scores.mean())

best_k = list(k_values)[np.argmax(cv_scores)]
print(f"Best k: {best_k}, CV accuracy: {max(cv_scores):.3f}")

# Important: always use odd k for binary classification to break ties

from sklearn.neighbors import KNeighborsClassifier
from sklearn.model_selection import cross_val_score
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler

for metric in ['euclidean', 'manhattan', 'cosine']:
    knn = make_pipeline(StandardScaler(),
                        KNeighborsClassifier(n_neighbors=5, metric=metric))
    score = cross_val_score(knn, X_train, y_train, cv=5).mean()
    print(f"{metric:12s}: {score:.3f}")

The Curse of Dimensionality

In high-dimensional spaces, all points become approximately equidistant from any query point — the concept of "nearest" loses meaning. This is the curse of dimensionality, and it makes KNN ineffective in very high dimensions (p >> 20) without dimensionality reduction.

import numpy as np

for d in [2, 10, 100, 1000]:
    # Sample 1000 random points in d dimensions
    X = np.random.randn(1000, d)
    query = np.zeros(d)
    dists = np.linalg.norm(X - query, axis=1)

    nearest = dists.min()
    farthest = dists.max()
    ratio = farthest / nearest
    print(f"d={d:4d}: min_dist={nearest:.2f}, max_dist={farthest:.2f}, ratio={ratio:.2f}")

# As d increases, max_dist/min_dist -> 1 (all points equally far)
# Solution: PCA or other dim reduction before KNN

Weighted KNN

from sklearn.neighbors import KNeighborsClassifier

# Uniform: each neighbour votes equally
knn_uniform = KNeighborsClassifier(n_neighbors=5, weights='uniform')

# Distance-weighted: closer neighbours vote more strongly
knn_dist = KNeighborsClassifier(n_neighbors=5, weights='distance')

for name, knn in [("uniform", knn_uniform), ("distance", knn_dist)]:
    knn.fit(X_train, y_train)
    print(f"{name}: {knn.score(X_test, y_test):.3f}")
