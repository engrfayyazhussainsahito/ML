Lesson 15.3: Self-Supervised Learning

Learning Without Labels

Self-supervised learning (SSL) creates supervision signals from the structure of unlabelled data itself. A pretext task (e.g. predict a masked word, predict rotation angle, or maximise similarity of augmented views) forces the model to learn rich representations. These representations transfer remarkably well to downstream tasks — often matching or exceeding fully supervised baselines with far fewer labelled examples.

Contrastive Learning: SimCLR

import torch, torch.nn as nn, torch.nn.functional as F
import torchvision.transforms as T

# Two random augmentations of the same image = positive pair
# Augmentations of different images = negative pairs
# Learn: same image -> nearby in embedding space; different -> far apart

class SimCLR(nn.Module):
    def __init__(self, encoder, projection_dim=128):
        super().__init__()
        self.encoder = encoder   # e.g. ResNet-50 without final layer
        self.projector = nn.Sequential(
            nn.Linear(2048, 512), nn.ReLU(),
            nn.Linear(512, projection_dim)
        )

    def forward(self, x):
        h = self.encoder(x)    # representations
        z = self.projector(h)  # projection for contrastive loss
        return h, F.normalize(z, dim=1)   # L2-normalise projections

def nt_xent_loss(z1, z2, temperature=0.5):
    """
    Normalised temperature-scaled cross entropy (NT-Xent).
    For each sample i, z1[i] and z2[i] should be most similar (positive pair).
    All other pairs are negatives.
    """
    N = z1.size(0)
    z = torch.cat([z1, z2], dim=0)          # (2N, d)
    sim = (z @ z.T) / temperature            # (2N, 2N) cosine similarity

    # Mask out self-similarity
    mask = torch.eye(2*N, dtype=bool)
    sim.masked_fill_(mask, -1e9)

    # Positive pairs: (i, i+N) and (i+N, i)
    pos_sim = torch.cat([torch.diag(sim, N), torch.diag(sim, -N)])  # (2N,)
    loss = -pos_sim + torch.logsumexp(sim, dim=1)
    return loss.mean()

# Data augmentation pipeline
augment = T.Compose([

Masked Autoencoders (MAE)

import torch, torch.nn as nn

class MAE(nn.Module):
    """
    Masked Autoencoder (He et al., 2021).
    Mask 75% of image patches, reconstruct from 25%.
    Forces learning of rich visual representations.
    """
    def __init__(self, patch_size=16, mask_ratio=0.75, d_model=768):
        super().__init__()
        self.patch_size  = patch_size
        self.mask_ratio  = mask_ratio
        # Encoder: ViT backbone, processes only visible patches
        # Decoder: lightweight Transformer, reconstructs all patches
        self.mask_token  = nn.Parameter(torch.zeros(1, 1, d_model))

    def random_masking(self, x):
        """Randomly mask a fraction of patches."""
        N, L, D = x.shape
        n_keep  = int(L * (1 - self.mask_ratio))
        noise   = torch.rand(N, L, device=x.device)
        ids_sort = noise.argsort(dim=1)
        ids_keep = ids_sort[:, :n_keep]
        x_masked = torch.gather(x, 1, ids_keep.unsqueeze(-1).expand(-1,-1,D))
        return x_masked, ids_sort, ids_keep

# Self-supervised pretraining objective: reconstruct pixel values of masked patches
# Loss: MSE on masked patches only
# After pretraining: throw away decoder, fine-tune encoder on downstream task
