Lesson 13.3: Diffusion Models

The Diffusion Principle

Diffusion models (Ho et al., 2020) are inspired by non-equilibrium thermodynamics. The forward process gradually adds Gaussian noise to data over T timesteps until it becomes pure noise. The model learns the reverse process: given noisy data at step t, predict the noise that was added, then remove it. At inference, start from pure noise and iteratively denoise.

The Forward Process

import torch
import numpy as np

# Noise schedule: how much noise is added at each timestep
T = 1000
beta_start, beta_end = 1e-4, 0.02
betas = torch.linspace(beta_start, beta_end, T)   # linear schedule

# Precompute cumulative products (used for efficient training)
alphas      = 1 - betas
alpha_bars  = torch.cumprod(alphas, dim=0)
sqrt_ab     = torch.sqrt(alpha_bars)
sqrt_1m_ab  = torch.sqrt(1 - alpha_bars)

def add_noise(x0, t, noise=None):
    """
    Forward process: q(x_t | x_0) = N(sqrt(alpha_bar_t)*x0, (1-alpha_bar_t)*I)
    Equivalent to: x_t = sqrt(ab_t)*x0 + sqrt(1-ab_t)*epsilon
    """
    if noise is None:
        noise = torch.randn_like(x0)
    sqrt_ab_t   = sqrt_ab[t].view(-1, *([1]*len(x0.shape[1:])))
    sqrt_1m_t   = sqrt_1m_ab[t].view(-1, *([1]*len(x0.shape[1:])))
    x_t = sqrt_ab_t * x0 + sqrt_1m_t * noise
    return x_t, noise   # return both x_t and the actual noise epsilon

# Example
x0 = torch.randn(4, 1, 28, 28)   # clean images
t  = torch.randint(0, T, (4,))

x_t, true_noise = add_noise(x0, t)
print(f"Noisy image at t={t}: {x_t.shape}")

# At t=0: x_t ≈ x0  (very little noise)
# At t=T: x_t ≈ N(0,I) (almost pure noise)
print(f"t=0  signal ratio: {sqrt_ab[0]:.3f}")
print(f"t=499 signal ratio: {sqrt_ab[499]:.3f}")
print(f"t=999 signal ratio: {sqrt_ab[999]:.4f}")

The U-Net Denoiser

import torch, torch.nn as nn

class SinusoidalTimeEmbedding(nn.Module):
    """Encode timestep t as a sinusoidal positional embedding."""
    def __init__(self, dim):
        super().__init__()
        self.dim = dim

    def forward(self, t):
        half = self.dim // 2
        freqs = torch.exp(-torch.arange(half, device=t.device) *
                          (np.log(10000) / (half - 1)))
        emb = t[:, None].float() * freqs[None]
        return torch.cat([emb.sin(), emb.cos()], dim=-1)

class ResBlock(nn.Module):
    """Residual block conditioned on time embedding."""
    def __init__(self, in_ch, out_ch, time_dim):
        super().__init__()
        self.conv1 = nn.Sequential(nn.GroupNorm(8, in_ch), nn.SiLU(),
                                   nn.Conv2d(in_ch, out_ch, 3, padding=1))
        self.time_mlp = nn.Sequential(nn.SiLU(), nn.Linear(time_dim, out_ch))
        self.conv2 = nn.Sequential(nn.GroupNorm(8, out_ch), nn.SiLU(),
                                   nn.Conv2d(out_ch, out_ch, 3, padding=1))
        self.shortcut = nn.Conv2d(in_ch, out_ch, 1) if in_ch != out_ch else nn.Identity()

    def forward(self, x, t_emb):
        h = self.conv1(x)
        h = h + self.time_mlp(t_emb)[:, :, None, None]
        h = self.conv2(h)
        return h + self.shortcut(x)

Training and Sampling

def train_step(model, x0, optimiser):
    """Simplified DDPM training step."""
    t = torch.randint(0, T, (x0.size(0),))
    x_t, noise = add_noise(x0, t)

    # Model predicts the noise
    predicted_noise = model(x_t, t)
    loss = nn.MSELoss()(predicted_noise, noise)

    optimiser.zero_grad(); loss.backward(); optimiser.step()
    return loss.item()

def ddpm_sample(model, shape, device='cpu'):
    """DDPM reverse process: start from noise, iteratively denoise."""
    x = torch.randn(shape, device=device)   # start from pure noise

    for t in reversed(range(T)):
        t_batch = torch.full((shape[0],), t, device=device, dtype=torch.long)
        predicted_noise = model(x, t_batch)

        alpha_t    = alphas[t]
        alpha_bar_t = alpha_bars[t]
        beta_t     = betas[t]

        # Reverse step: remove predicted noise
        x = (1 / torch.sqrt(alpha_t)) * (
            x - (beta_t / torch.sqrt(1 - alpha_bar_t)) * predicted_noise
        )
        if t > 0:
            x += torch.sqrt(beta_t) * torch.randn_like(x)

    return x.clamp(-1, 1)
